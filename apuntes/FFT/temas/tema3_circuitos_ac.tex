\chapter{Fundamentos de Teoría de Circuitos: Corriente Alterna (AC)}

\section{Señales Alternas Sinusoidales y Representación Fasorial}

En este tema estudiamos la resolución de circuitos excitados por fuentes de señal variable en el tiempo en \textbf{régimen permanente sinusoidal}. Las señales más empleadas en electrotecnia, electrónica y telecomunicaciones son las ondas periódicas senoidales (seno o coseno de periodo $2\pi$ radianes).

\subsection{Caracterización de una Señal Periódica Senoidal}

Una magnitud eléctrica alterna sinusoidal (tensión o intensidad) se expresa en función del tiempo como:
\begin{equation}
    x(t) = X_m \cos(\omega t + \phi)
\end{equation}

Sus parámetros característicos fundamentales son:
\begin{itemize}
    \item \textbf{Amplitud o valor máximo ($X_m$)}: Valor de pico que alcanza la señal. El valor pico a pico es $X_{pp} = 2 X_m$.
    \item \textbf{Periodo ($T$)}: Tiempo que tarda la señal en repetirse íntegramente ($\si{\second}$).
    \item \textbf{Frecuencia lineal ($f$)}: Número de ciclos completos por segundo:
    \begin{equation}
        f = \frac{1}{T} \qquad [\text{SI: Hercio, }\si{\hertz} = \si{\second^{-1}}]
    \end{equation}
    \item \textbf{Frecuencia angular o pulsación ($\omega$)}: Velocidad de variación angular en radianes por segundo:
    \begin{equation}
        \omega T = 2\pi \implies \omega = \frac{2\pi}{T} = 2\pi f \qquad [\si{\radian/\second}]
    \end{equation}
    \item \textbf{Fase inicial en el origen ($\phi$)}: Representa la fracción del periodo en la cual la señal está desplazada (adelantada si $\phi > 0$ o retrasada si $\phi < 0$) respecto al origen de tiempos $t = 0$. Si el primer máximo ocurre un tiempo $t'$ antes del origen ($t = -t'$), se verifica:
    \begin{equation}
        \phi = \omega t' \implies \frac{t'}{T} = \frac{\phi}{2\pi}
    \end{equation}
    \item \textbf{Valor eficaz o RMS ($X_{\text{ef}}$ o $X_{\text{rms}}$)}: Es el valor de una corriente continua que disiparía la misma potencia media sobre una resistencia pura. Para una señal sinusoidal pura:
    \begin{equation}
        X_{\text{ef}} = \sqrt{\frac{1}{T}\int_0^T x^2(t)\,dt} = \frac{X_m}{\sqrt{2}} \approx 0{,}707\,X_m
    \end{equation}
\end{itemize}

\begin{aviso}[Identidades trigonométricas para unificar senos y cosenos]
Para operar con fasores en un mismo circuito, \textbf{todas las fuentes deben estar expresadas con la misma función trigonométrica} (habitualmente coseno). Recuerda las relaciones de conversión:
\begin{equation}
    \sin(\alpha) = \cos\left(\alpha - \frac{\pi}{2}\right) = \cos(\alpha - 90^\circ), \qquad -\sin(\alpha) = \cos\left(\alpha + \frac{\pi}{2}\right) = \cos(\alpha + 90^\circ)
\end{equation}
\end{aviso}

\subsection{Repaso de Números Complejos para Circuitos AC}

A la hora de trabajar en corriente alterna no operamos resolviendo ecuaciones diferenciales con funciones senoidales en el tiempo, sino con sus \textbf{fasores} asociados, que son números complejos $\mathbf{Z}$ con parte real $a$ y parte imaginaria $b$ (en electrotecnia se emplea la letra $j = \sqrt{-1}$, con $j^2 = -1$, para no confundirla con la intensidad $i$).

\subsubsection{1. Las 4 Formas de Expresar un Número Complejo}
\begin{equation}
    \mathbf{Z} = \underbrace{a + j b}_{\text{Binomial}} = \underbrace{(a,\,b)}_{\text{Cartesiana}} = \underbrace{r\,e^{j\varphi}}_{\text{Exponencial (Euler)}} = \underbrace{r\angle\varphi}_{\text{Polar (Módulo y Fase)}}
\end{equation}

\begin{ley}[Conversión entre Forma Binomial y Forma Polar]
\begin{itemize}
    \item \textbf{Paso de Binomial ($a + jb$) a Polar ($r\angle\varphi$)}:
    \begin{equation}
        \text{Módulo: } r = |\mathbf{Z}| = \sqrt{a^2 + b^2}, \qquad \text{Fase: } \varphi = \begin{cases} \arctan\left(\dfrac{b}{a}\right) & \text{si } a > 0 \\[8pt] \arctan\left(\dfrac{b}{a}\right) + \pi\ (\text{o } +180^\circ) & \text{si } a < 0 \end{cases}
    \end{equation}
    \item \textbf{Paso de Polar ($r\angle\varphi$) a Binomial ($a + jb$)}:
    \begin{equation}
        a = r\cos\varphi, \qquad b = r\sin\varphi \implies \mathbf{Z} = r\cos\varphi + j\,r\sin\varphi
    \end{equation}
\end{itemize}
\end{ley}

\begin{ejemplo}[Paso de binomial a polar en el segundo cuadrante ($a < 0$)]
Dado el número complejo $\mathbf{Z} = -3 + 4j$, halla su expresión en forma polar:
\begin{itemize}
    \item \textbf{Módulo}: $r = |\mathbf{Z}| = \sqrt{(-3)^2 + 4^2} = \sqrt{9 + 16} = \sqrt{25} = 5$.
    \item \textbf{Fase}: Como la parte real es negativa ($a = -3 < 0$, segundo cuadrante), debemos sumar $\pi\,\si{\radian}$ ($180^\circ$) al resultado de la calculadora:
    \begin{equation*}
        \varphi = \arctan\left(\frac{4}{-3}\right) + 180^\circ = -53{,}13^\circ + 180^\circ = 126{,}87^\circ \implies \boxed{\mathbf{Z} = 5\angle 126{,}87^\circ}
    \end{equation*}
\end{itemize}
\end{ejemplo}

\subsubsection{2. Conjugado y Operaciones con Números Complejos}
\begin{itemize}
    \item \textbf{Complejo conjugado ($\mathbf{Z}^*$)}: Es el mismo número complejo pero con la parte imaginaria (o el ángulo de fase) cambiada de signo: si $\mathbf{Z} = a + jb = r\angle\varphi$, entonces $\mathbf{Z}^* = a - jb = r\angle -\varphi$. Se cumple siempre que el producto de un complejo por su conjugado es un número real puro igual al cuadrado de su módulo:
    \begin{equation}
        \mathbf{Z}\cdot\mathbf{Z}^* = (a + jb)(a - jb) = a^2 - j^2 b^2 = a^2 + b^2 = |\mathbf{Z}|^2
    \end{equation}
    \item \textbf{Suma y resta} (más sencillas en \textbf{forma binomial}):
    \begin{equation}
        \mathbf{Z}_1 \pm \mathbf{Z}_2 = (a + jb) \pm (c + jd) = (a \pm c) + j(b \pm d)
    \end{equation}
    \item \textbf{Multiplicación} (más sencilla en \textbf{forma polar o exponencial}):
    \begin{equation}
        \mathbf{Z}_1 \cdot \mathbf{Z}_2 = (r_1\angle\varphi_1)\cdot(r_2\angle\varphi_2) = (r_1 r_2)\angle(\varphi_1 + \varphi_2) = r_1 r_2\,e^{j(\varphi_1 + \varphi_2)}
    \end{equation}
    En forma binomial: $(a+jb)(c+jd) = (ac - bd) + j(ad + bc)$.
    \item \textbf{División} (en forma polar se dividen módulos y se restan fases; en forma binomial se multiplica numerador y denominador por el \textbf{conjugado del denominador}):
    \begin{equation}
        \frac{\mathbf{Z}_1}{\mathbf{Z}_2} = \frac{r_1\angle\varphi_1}{r_2\angle\varphi_2} = \frac{r_1}{r_2}\angle(\varphi_1 - \varphi_2) = \frac{(a+jb)(c-jd)}{c^2 + d^2} = \left(\frac{ac + bd}{c^2 + d^2}\right) + j\left(\frac{bc - ad}{c^2 + d^2}\right)
    \end{equation}
\end{itemize}

\subsection{El Fasor, la Impedancia Compleja ($\mathbf{Z}$) y la Admitancia ($\mathbf{Y}$)}

Un \textbf{fasor} es el número complejo cuyo módulo es la amplitud ($V_m$ o $I_m$) y cuyo argumento es la fase en el origen ($\phi$) de una señal alterna sinusoidal de pulsación $\omega$:
\begin{equation}
    v(t) = V_m\cos(\omega t + \phi) \quad \longleftrightarrow \quad \mathbf{V} = V_m\angle\phi = V_m\,e^{j\phi} = V_m\cos\phi + j\,V_m\sin\phi
\end{equation}

\begin{definicion}[Impedancia ($\mathbf{Z}$) y Admitancia ($\mathbf{Y}$)]
\begin{itemize}
    \item La \textbf{impedancia compleja} $\mathbf{Z}$ de un elemento pasivo es el cociente entre el fasor de tensión $\mathbf{V}$ en sus extremos y el fasor de intensidad $\mathbf{I}$ que lo atraviesa (Ley de Ohm generalizada en CA):
    \begin{equation}
        \boxed{\mathbf{Z} = \frac{\mathbf{V}}{\mathbf{I}} = R + j X = |\mathbf{Z}|\angle\theta} \qquad [\si{\ohm}]
    \end{equation}
    donde $R = \text{Re}(\mathbf{Z})$ es la \textbf{parte resistiva} (resistencia) y $X = \text{Im}(\mathbf{Z})$ es la \textbf{parte reactiva} (reactancia).
    \item La \textbf{admitancia compleja} $\mathbf{Y}$ es la inversa de la impedancia:
    \begin{equation}
        \boxed{\mathbf{Y} = \frac{1}{\mathbf{Z}} = \frac{\mathbf{I}}{\mathbf{V}} = G + j B} \qquad [\text{Siemens, }\si{\siemens} = \si{\ohm^{-1}}]
    \end{equation}
\end{itemize}
\end{definicion}

\subsection{Relación Tensión-Intensidad en los Elementos Pasivos en CA}

\begin{enumerate}
    \item \textbf{Resistencia ideal ($R$)}:
    \begin{equation}
        \mathbf{Z}_R = R = R\angle 0^\circ, \qquad \mathbf{Y}_R = \frac{1}{R}
    \end{equation}
    Si $\mathbf{I}_R = I_m\angle\phi$, entonces $\mathbf{V}_R = \mathbf{I}_R \mathbf{Z}_R = (I_m R)\angle(\phi + 0^\circ)$. En una resistencia, \textbf{la tensión y la intensidad tienen exactamente la misma fase} (están en fase): si $i_R(t) = I_m\cos(\omega t + \phi)$, entonces $v_R(t) = I_m R\cos(\omega t + \phi)$.

    \item \textbf{Condensador ideal ($C$)}:
    \begin{equation}
        \mathbf{Z}_C = \frac{1}{j\omega C} = \frac{-j}{\omega C} = \frac{1}{\omega C}\angle -90^\circ, \qquad \mathbf{Y}_C = j\omega C
    \end{equation}
    Si $\mathbf{I}_C = I_m\angle\phi$, entonces $\mathbf{V}_C = \mathbf{I}_C \mathbf{Z}_C = \frac{I_m}{\omega C}\angle(\phi - 90^\circ)$. En un condensador, \textbf{la tensión va retrasada $90^\circ$ ($\pi/2\,\si{\radian}$) respecto a la intensidad} (o equivalentemente, la intensidad adelanta $\pi/2\,\si{\radian}$ a la tensión): si $v_C(t) = V_m\cos(\omega t + \phi)$, entonces $i_C(t) = V_m C\omega\cos\left(\omega t + \phi + \frac{\pi}{2}\right)$.

    \item \textbf{Bobina o inductor ideal ($L$)}:
    \begin{equation}
        \mathbf{Z}_L = j\omega L = \omega L\angle +90^\circ, \qquad \mathbf{Y}_L = \frac{1}{j\omega L} = \frac{-j}{\omega L}
    \end{equation}
    Si $\mathbf{I}_L = I_m\angle\phi$, entonces $\mathbf{V}_L = \mathbf{I}_L \mathbf{Z}_L = (\omega L I_m)\angle(\phi + 90^\circ)$. En una bobina, \textbf{la tensión va adelantada $90^\circ$ ($\pi/2\,\si{\radian}$) respecto a la intensidad}: si $i_L(t) = I_m\cos(\omega t + \phi)$, entonces $v_L(t) = I_m\omega L\cos\left(\omega t + \phi + \frac{\pi}{2}\right)$.
\end{enumerate}

\subsection{Análisis Completo del Circuito RC Serie en CA}

Al trabajar con impedancias complejas es como si todos los elementos fueran resistivos y cumplen la Ley de Ohm y todas las reglas de asociación serie/paralelo, divisores, Kirchhoff, Mallas, Nudos, Superposición, Thévenin y Norton. Analicemos en detalle el circuito RC serie excitado por $v_s(t) = |\mathbf{V}_s|\cos(\omega t + \phi)$ ($\mathbf{V}_s = |\mathbf{V}_s|\angle\phi$):

\begin{enumerate}
    \item \textbf{Ecuación de malla e intensidad fasorial}: Con $\mathbf{Z}_R = R$ y $\mathbf{Z}_C = \frac{1}{j\omega C}$:
    \begin{equation}
        \mathbf{V}_s = \mathbf{I}(\mathbf{Z}_R + \mathbf{Z}_C) = \mathbf{I}\left(R + \frac{1}{j\omega C}\right) = \mathbf{I}\left(\frac{1 + j\omega C R}{j\omega C}\right) \implies \mathbf{I} = \frac{j\omega C}{1 + j\omega C R}\,\mathbf{V}_s
    \end{equation}
    \item \textbf{Caída de tensión en el condensador ($\mathbf{V}_C$)}:
    \begin{equation}
        \mathbf{V}_C = \mathbf{I}\cdot\mathbf{Z}_C = \frac{j\omega C\,\mathbf{V}_s}{1 + j\omega C R}\cdot\frac{1}{j\omega C} = \frac{1}{1 + j\omega C R}\,\mathbf{V}_s
    \end{equation}
    Racionalizando (multiplicando arriba y abajo por el conjugado $1 - j\omega C R$):
    \begin{equation}
        \mathbf{V}_C = \frac{\mathbf{V}_s(1 - j\omega C R)}{(1 + j\omega C R)(1 - j\omega C R)} = \frac{\mathbf{V}_s}{1 + (\omega C R)^2}\,(1 - j\omega C R)
    \end{equation}
    \item \textbf{Módulo, fase y expresión en el dominio del tiempo}:
    \begin{equation}
        |\mathbf{V}_C| = \frac{|\mathbf{V}_s|}{\sqrt{1 + (\omega C R)^2}}, \qquad \varphi_C = \phi - \arctan(\omega C R) \implies \mathbf{V}_C = \frac{|\mathbf{V}_s|}{\sqrt{1 + (\omega C R)^2}}\angle\big(\phi - \arctan(\omega C R)\big)
    \end{equation}
    Pasando al dominio del tiempo:
    \begin{equation}
        \boxed{v_C(t) = \frac{|\mathbf{V}_s|}{\sqrt{1 + (\omega C R)^2}}\,\cos\big[\omega t + \phi - \arctan(\omega C R)\big]}
    \end{equation}
\end{enumerate}

\section{Función de Transferencia, Diagramas de Bode y Filtros Pasivos}

\subsection{Función de Transferencia $\mathbf{T}(\omega)$ y Frecuencia de Corte $\omega_0$}

\begin{definicion}[Función de Transferencia y Frecuencia de Corte]
\begin{itemize}
    \item La \textbf{función de transferencia} $\mathbf{T}(\omega)$ (o $\mathbf{H}(j\omega)$) es la relación entre el fasor de tensión de salida $\mathbf{V}_o(\omega)$ y el fasor de tensión de entrada $\mathbf{V}_i(\omega)$ en función de la frecuencia angular $\omega$:
    \begin{equation}
        \boxed{\mathbf{T}(\omega) = \frac{\mathbf{V}_o(\omega)}{\mathbf{V}_i(\omega)} = |\mathbf{T}(\omega)|\,\angle\text{Fase}\{\mathbf{T}(\omega)\}}
    \end{equation}
    \item La \textbf{frecuencia de corte} $\omega_0$ (o $\omega_c$) es aquella frecuencia angular para la cual el módulo de la función de transferencia cae a $\frac{1}{\sqrt{2}} \approx 0{,}707$ de su valor máximo (lo que equivale a una caída de $-3\,\si{\decibel}$ en amplitud o a la mitad de la potencia):
    \begin{equation}
        |\mathbf{T}(\omega_0)| = \frac{|\mathbf{T}|_{\max}}{\sqrt{2}} \approx 0{,}707\,|\mathbf{T}|_{\max} \qquad \left(\text{En un filtro RC: } \boxed{\omega_0 = \frac{1}{R C}}, \quad \text{En un filtro RL: } \boxed{\omega_0 = \frac{R}{L}}\right)
    \end{equation}
\end{itemize}
\end{definicion}

\subsection{Diagramas de Bode de Amplitud y de Fase}

El \textbf{Diagrama de Bode} sirve para analizar y representar gráficamente el comportamiento en frecuencia de un circuito mediante dos gráficas frente a $\log_{10}(\omega)$ (o poniendo el eje $\omega$ en escala logarítmica, donde cada \textbf{década} multiplica la frecuencia por $10$, p. ej. de $100\,\si{\hertz}$ a $1000\,\si{\hertz}$):
\begin{enumerate}
    \item \textbf{Diagrama de Bode de amplitud}: Representa la ganancia en \textbf{decibelios ($\si{\decibel}$)}, definida como:
    \begin{equation}
        |\mathbf{T}(\omega)|_{\si{\decibel}} = 20\log_{10}|\mathbf{T}(\omega)| \qquad [\si{\decibel}]
    \end{equation}
    \item \textbf{Diagrama de Bode de fase}: Representa el desfase $\text{Fase}\{\mathbf{T}(\omega)\}$ (en radianes o grados) frente a $\log_{10}(\omega)$.
\end{enumerate}

\subsection{Filtro RC Paso Bajo (Salida en el Condensador: $\mathbf{V}_o = \mathbf{V}_C$)}
Tomando como salida la tensión en el condensador ($\mathbf{V}_o = \mathbf{V}_C$) y como entrada la fuente ($\mathbf{V}_i = \mathbf{V}_s$), con $\omega_0 = \frac{1}{RC}$:
\begin{equation}
    \mathbf{T}(\omega) = \frac{1}{1 + j\omega R C} = \frac{1}{\sqrt{1 + (\omega R C)^2}}\,\angle\big(-\arctan(\omega R C)\big) = \frac{1}{\sqrt{1 + (\omega/\omega_0)^2}}\,\angle\left(-\arctan\frac{\omega}{\omega_0}\right)
\end{equation}
\begin{itemize}
    \item \textbf{Para frecuencias bajas ($\omega \ll \omega_0$)}: $|\mathbf{T}(\omega)| \approx 1 \implies 20\log_{10}(1) = 0\,\si{\decibel}$, y $\text{Fase}\{\mathbf{T}(\omega)\} \approx -\arctan(0) = 0\,\si{\radian}$. Por tanto, para frecuencias bajas la señal de salida $v_o(t)$ y la de entrada $v_i(t)$ tienen \textbf{la misma amplitud y la misma fase} (pasan sin atenuación).
    \item \textbf{En la frecuencia de corte ($\omega = \omega_0 = \frac{1}{RC}$)}: $|\mathbf{T}(\omega_0)| = \frac{1}{\sqrt{2}} \approx 0{,}707 \implies 20\log_{10}(1/\sqrt{2}) = -3\,\si{\decibel}$, y $\text{Fase}\{\mathbf{T}(\omega_0)\} = -\arctan(1) = -\frac{\pi}{4}\,\si{\radian} = -45^\circ$.
    \item \textbf{Para frecuencias altas ($\omega \gg \omega_0$)}: $|\mathbf{T}(\omega)| \approx \frac{\omega_0}{\omega} \implies 20\log_{10}\left(\frac{\omega_0}{\omega}\right)$, que cae con una pendiente asintótica de \textbf{$-20\,\si{\decibel}/\text{década}$}, y $\text{Fase}\{\mathbf{T}(\omega)\} \approx -\arctan(\infty) = -\frac{\pi}{2}\,\si{\radian} = -90^\circ$. A altas frecuencias la amplitud de salida es mucho menor que la de entrada y va retrasada $\pi/2$.
\end{itemize}

\section{Potencia en Corriente Alterna}

\begin{aviso}[Regla Fundamental para la Potencia en CA]
Para calcular la potencia instantánea en corriente alterna \textbf{debemos usar siempre las expresiones de tensión e intensidad en el dominio del tiempo}:
\begin{equation}
    p(t) = v(t) \cdot i(t)
\end{equation}
\textbf{No se pueden multiplicar directamente los fasores $\mathbf{V}\cdot\mathbf{I}$} sin conjugar, ya que el producto directo de dos fasores no coincide con el fasor ni con el valor medio de la potencia.
\end{aviso}

\subsection{Potencia Instantánea, Potencia Activa (Media) y Potencia Reactiva}

Sean la tensión y la intensidad en un dipolo:
\begin{equation}
    v(t) = V_m\cos(\omega t + \theta_v), \qquad i(t) = I_m\cos(\omega t + \theta_i)
\end{equation}
Aplicando la identidad trigonométrica del producto de cosenos $\cos A\cdot\cos B = \frac{1}{2}\big[\cos(A+B) + \cos(A-B)\big]$, la \textbf{potencia instantánea} resulta:
\begin{equation}
    \boxed{p(t) = V_m\cos(\omega t + \theta_v)\cdot I_m\cos(\omega t + \theta_i) = \frac{V_m\,I_m}{2}\,\Big[\underbrace{\cos(2\omega t + \theta_v + \theta_i)}_{\substack{\text{Potencia reactiva}\\\text{o fluctuante (media nula)}}} + \underbrace{\cos(\theta_v - \theta_i)}_{\substack{\text{Potencia activa}\\\text{(constante)}}}\Big]}
\end{equation}

Al promediar a lo largo de un periodo completo $T$, el término cosenoidal de frecuencia doble $2\omega$ tiene media nula, por lo que la \textbf{potencia media o potencia activa} es:
\begin{equation}
    \boxed{P_{\text{media}} = P = \frac{V_m\,I_m}{2}\,\cos(\theta_v - \theta_i) = V_{\text{ef}}\,I_{\text{ef}}\,\cos\varphi} \qquad [\text{Vatio, }\si{\watt}]
\end{equation}
donde $\varphi = \theta_v - \theta_i$ es el ángulo de desfase entre tensión e intensidad y $\text{fp} = \cos\varphi$ es el \textbf{factor de potencia}.
\begin{itemize}
    \item \textbf{En una resistencia pura} ($\theta_v = \theta_i \implies \varphi = 0^\circ$): $\cos(0^\circ) = 1 \implies P_R = \frac{V_m I_m}{2} = \frac{1}{2}I_m^2 R > 0$.
    \item \textbf{En una bobina o condensador ideal} ($\theta_v - \theta_i = \pm 90^\circ = \pm \frac{\pi}{2}$): $\cos(\pm\pi/2) = 0 \implies \boxed{P_{L,\text{media}} = P_{C,\text{media}} = 0\,\si{\watt}}$. Los elementos reactivos puros no disipan potencia media: la absorben durante un cuarto de ciclo y la devuelven íntegramente en el siguiente.
\end{itemize}

En ingeniería eléctrica se definen además la \textbf{potencia reactiva} $Q = \frac{V_m I_m}{2}\sin(\theta_v - \theta_i) = V_{\text{ef}}I_{\text{ef}}\sin\varphi$ $[\si{\volt\ampere\text{r}}]$ y la \textbf{potencia aparente} $S = \frac{V_m I_m}{2} = V_{\text{ef}}I_{\text{ef}} = \sqrt{P^2 + Q^2}$ $[\si{\volt\ampere}]$, que forman el \textbf{triángulo de potencias}:

\begin{ejemplo}[Análisis completo de un circuito RLC serie (pág. 65)]
En el circuito RLC en serie de la figura, la fuente de tensión alterna tiene una amplitud de $100\,\si{\volt}$, frecuencia $f = 79{,}6\,\si{\hertz}$ y fase inicial nula, siendo $R = 100\,\si{\ohm}$, $L = 1\,\si{\henry}$ y $C = 5\,\si{\micro\farad}$. Calcula las expresiones temporales de la intensidad $i(t)$ y de las caídas de tensión en cada elemento: $v_R(t)$, $v_L(t)$ y $v_C(t)$.

\begin{center}
\begin{circuitikz}[scale=0.92, transform shape]
    \draw (0,0) to[sV, l={$100\,\si{\volt},\,79{,}6\,\si{\hertz}$}] (0,2.2)
          to[R, l={$R=100\,\si{\ohm}$}] (2.6,2.2)
          to[L, l={$L=1\,\si{\henry}$}] (5.2,2.2)
          to[C, l={$C=5\,\si{\micro\farad}$}] (5.2,0) -- (0,0);
    \draw[-{Stealth}, NavyBlue, thick] (2.2,1.1) arc (150:-150:0.45) node[midway, left] {$\mathbf{I}$};
\end{circuitikz}
\end{center}

\textbf{Resolución paso a paso}:
\begin{enumerate}
    \item \textbf{Cálculo de la pulsación $\omega$ y expresión temporal de la fuente}:
    \begin{equation*}
        \omega = 2\pi f = 2\pi \cdot 79{,}6\,\si{\hertz} = 500\,\si{\radian/\second} \implies v(t) = 100\cos(500 t + 0)\,\si{\volt}
    \end{equation*}
    \item \textbf{Impedancias de cada elemento y su equivalente serie en forma binomial y polar}:
    \begin{align*}
        \mathbf{Z}_R &= 100\,\si{\ohm} = 100\angle 0^\circ\,\si{\ohm} \\
        \mathbf{Z}_C &= \frac{1}{j\omega C} = \frac{-j}{500 \cdot 5\times 10^{-6}} = -j 400\,\si{\ohm} = 400\angle -90^\circ\,\si{\ohm} \\
        \mathbf{Z}_L &= j\omega L = j\,500\cdot 1 = j 500\,\si{\ohm} = 500\angle 90^\circ\,\si{\ohm} \\
        \mathbf{Z}_{eq} &= \mathbf{Z}_R + \mathbf{Z}_L + \mathbf{Z}_C = 100 + j 500 - j 400 = (100 + j 100)\,\si{\ohm}
    \end{align*}
    Pasando $\mathbf{Z}_{eq}$ a forma polar:
    \begin{equation*}
        |\mathbf{Z}_{eq}| = \sqrt{100^2 + 100^2} = 141{,}4\,\si{\ohm}, \quad \text{fase} = \arctan\left(\frac{100}{100}\right) = 45^\circ \implies \mathbf{Z}_{eq} = 141{,}4\angle 45^\circ\,\si{\ohm}
    \end{equation*}
    \item \textbf{Fasor de la fuente e intensidad de la malla}:
    \begin{equation*}
        \mathbf{V} = 100\angle 0^\circ\,\si{\volt} \implies \mathbf{I} = \frac{\mathbf{V}}{\mathbf{Z}_{eq}} = \frac{100\angle 0^\circ}{141{,}4\angle 45^\circ} = 0{,}707\angle -45^\circ\,\si{\ampere} \implies \boxed{i(t) = 0{,}707\cos\left(500 t - \frac{\pi}{4}\right)\,\si{\ampere}}
    \end{equation*}
    \item \textbf{Caídas de tensión en cada elemento (en fasor y en el dominio del tiempo)}:
    \begin{itemize}
        \item \textbf{Resistencia}: $\mathbf{V}_R = \mathbf{I}\cdot\mathbf{Z}_R = (0{,}707\angle -45^\circ)\cdot(100\angle 0^\circ) = 70{,}7\angle -45^\circ\,\si{\volt}$
        \begin{equation*}
            \boxed{v_R(t) = 70{,}7\cos\left(500 t - \frac{\pi}{4}\right)\,\si{\volt}}
        \end{equation*}
        \item \textbf{Bobina}: $\mathbf{V}_L = \mathbf{I}\cdot\mathbf{Z}_L = (0{,}707\angle -45^\circ)\cdot(500\angle 90^\circ) = 353{,}5\angle 45^\circ\,\si{\volt}$
        \begin{equation*}
            \boxed{v_L(t) = 353{,}5\cos\left(500 t + \frac{\pi}{4}\right)\,\si{\volt}}
        \end{equation*}
        \item \textbf{Condensador}: $\mathbf{V}_C = \mathbf{I}\cdot\mathbf{Z}_C = (0{,}707\angle -45^\circ)\cdot(400\angle -90^\circ) = 282{,}8\angle -135^\circ\,\si{\volt}$
        \begin{equation*}
            \boxed{v_C(t) = 282{,}8\cos\left(500 t - \frac{3\pi}{4}\right)\,\si{\volt}}
        \end{equation*}
    \end{itemize}
\end{enumerate}
\end{ejemplo}

\begin{ejemplo}[Cálculo de potencias instantáneas y medias con dos fuentes (pág. 66)]
En el circuito de la figura se tienen dos fuentes $v_1(t) = 8\cos(\omega t)\,\si{\volt}$ y $v_2(t) = 6\cos\left(\omega t + \frac{\pi}{2}\right)\,\si{\volt}$ con $\omega = 200\,\si{\radian/\second}$, una resistencia $R = 600\,\si{\ohm}$ y una bobina $L = 4\,\si{\henry}$. Calcula la corriente $i(t)$ y las potencias instantáneas y medias en ambas fuentes y en la resistencia.

\begin{center}
\begin{circuitikz}[scale=0.92, transform shape]
    \draw (0,0) to[sV, l=$v_1(t)$] (0,2.2)
          to[R, l={$600\,\si{\ohm}$}] (3.2,2.2)
          to[L, l={$4\,\si{\henry}$}] (3.2,1.1)
          to[sV, l=$v_2(t)$] (3.2,0) -- (0,0);
    \draw[-{Stealth}, NavyBlue, thick] (1.3,1.1) arc (150:-150:0.42) node[midway, left] {$i$};
\end{circuitikz}
\end{center}

\textbf{Resolución paso a paso}:
\begin{enumerate}
    \item \textbf{Impedancias y fasores de las fuentes}:
    \begin{align*}
        \mathbf{Z}_R &= 600\,\si{\ohm}, \qquad \mathbf{Z}_L = j\omega L = j\,200\cdot 4 = j 800\,\si{\ohm} \\
        \mathbf{V}_1 &= 8\angle 0^\circ = 8 + j 0\,\si{\volt}, \qquad \mathbf{V}_2 = 6\angle 90^\circ = j 6\,\si{\volt}
    \end{align*}
    \item \textbf{Ecuación de malla por LTK}:
    \begin{equation*}
        \mathbf{V}_1 - \mathbf{V}_2 = \mathbf{I}(\mathbf{Z}_R + \mathbf{Z}_L) \implies 8 - j 6 = \mathbf{I}(600 + j 800) \implies \mathbf{I} = \frac{8 - j 6}{600 + j 800} = -j 0{,}01 = 0{,}01\angle -90^\circ\,\si{\ampere}
    \end{equation*}
    Pasando al dominio del tiempo:
    \begin{equation*}
        \boxed{i(t) = 0{,}01\cos\left(\omega t - \frac{\pi}{2}\right)\,\si{\ampere}}
    \end{equation*}
    \item \textbf{Cálculo de potencias instantáneas y medias}:
    \begin{itemize}
        \item \textbf{Fuente 1}:
        \begin{equation*}
            p_{V1}(t) = v_1(t)\cdot i(t) = 8\cos(\omega t)\cdot 0{,}01\cos\left(\omega t - \frac{\pi}{2}\right) = \frac{0{,}08}{2}\left[\cos\left(2\omega t - \frac{\pi}{2}\right) + \cos\left(\frac{\pi}{2}\right)\right]\,\si{\watt}
        \end{equation*}
        \begin{equation*}
            \boxed{P_{V1,\text{media}} = 0{,}04\cos\left(\frac{\pi}{2}\right) = 0\,\si{\watt}}
        \end{equation*}
        \item \textbf{Fuente 2} (la corriente $i(t)$ entra por el terminal positivo de $v_2$, luego lleva signo menos):
        \begin{equation*}
            p_{V2}(t) = -v_2(t)\cdot i(t) = -6\cos\left(\omega t + \frac{\pi}{2}\right)\cdot 0{,}01\cos\left(\omega t - \frac{\pi}{2}\right) = -\frac{0{,}06}{2}\big[\cos(2\omega t) + \cos(\pi)\big]\,\si{\watt}
        \end{equation*}
        Como $\cos(\pi) = -1$:
        \begin{equation*}
            \boxed{P_{V2,\text{media}} = -0{,}03\cdot(-1) = +0{,}03\,\si{\watt}} \quad (\text{suministra } 30\,\si{\milli\watt})
        \end{equation*}
        \item \textbf{Resistencia} ($v_R(t) = R\cdot i(t) = 6\cos(\omega t - \pi/2)\,\si{\volt}$):
        \begin{equation*}
            p_R(t) = v_R(t)\cdot i(t) = \frac{6\cdot 0{,}01}{2}\big[\cos(2\omega t - \pi) + \cos(0)\big]\,\si{\watt} \implies \boxed{P_{R,\text{media}} = 0{,}03\,\si{\watt}}
        \end{equation*}
    \end{itemize}
\end{enumerate}
\end{ejemplo}

\begin{ejercicio}[Ejercicios cortos de fasores e identificación de elementos (Ejercicios 4, 8, 9 y 10)]
\begin{enumerate}
    \item \textbf{Ejercicio 4}: \emph{¿Cuál es el valor máximo y el periodo de la onda $v(t) = 100\sin(377 t)\,\si{\volt}$?}
    \begin{itemize}
        \item \textbf{Valor máximo (amplitud)}: $V_m = \boxed{100\,\si{\volt}}$ (valor cuando el seno vale $1$).
        \item \textbf{Periodo}: Como $\omega = 377\,\si{\radian/\second}$:
        \begin{equation*}
            \boxed{T = \frac{2\pi}{\omega} = \frac{2\pi\,\si{\radian}}{377\,\si{\radian/\second}} \approx 0{,}0167\,\si{\second} = 16{,}7\,\si{\milli\second}}
        \end{equation*}
    \end{itemize}

    Sustituyendo $\mathbf{I}_2$ en la primera ecuación:
    \begin{equation*}
        4 - j 4 = (200 - j 400)\mathbf{I}_1 + j 400\left(-\frac{j}{2}\,\mathbf{I}_1\right) = (400 - j 400)\mathbf{I}_1 \implies \boxed{\mathbf{I}_1 = \mathbf{I}_{L1} = 0{,}01\,\si{\ampere} = 0{,}01\angle 0^\circ\,\si{\ampere}}
    \end{equation*}
    \begin{equation*}
        \boxed{\mathbf{I}_2 = \mathbf{I}_{L2} = -\frac{j}{2}\cdot 0{,}01 = -j\,5\times 10^{-3}\,\si{\ampere} = 5\times 10^{-3}\angle -90^\circ\,\si{\ampere}}
    \end{equation*}
    \item \textbf{Expresiones temporales de las corrientes por las bobinas}:
    \begin{equation*}
        \boxed{i_{L1}(t) = 0{,}01\cos(10^4 t)\,\si{\ampere}}, \qquad \boxed{i_{L2}(t) = 5\times 10^{-3}\cos\left(10^4 t - \frac{\pi}{2}\right)\,\si{\ampere}}
    \end{equation*}
\end{enumerate}
\end{ejercicio}

\begin{ejercicio}[Análisis por Nudos y Balance Completo de Potencias en CA (Ejercicio 31)]
Determina las ecuaciones temporales de las tensiones e intensidades de rama y las potencias instantáneas y medias en todos los elementos del siguiente circuito:

\begin{center}
\begin{circuitikz}[scale=0.9, transform shape]
    \draw (0,0) to[C, l={$-j 2\,\si{\ohm}$}, i<_=$I_4$] (0,2.4) node[above left] {$V_B$}
          to[R, l={$R_1=4\,\si{\ohm}$}] (2.4,2.4) -- (5.6,2.4);
    \node[above] at (4.0,2.4) {$V_A$};
    \draw (2.4,2.4) to[L, l={$j 1\,\si{\ohm}$}, i>^=$I_3$] (2.4,0);
    \draw (4.0,0) to[isource, l={$4\angle 0^\circ\,\si{\ampere}$}, i>^=$I_1$] (4.0,2.4);
    \draw (5.6,2.4) to[R, l={$R_2=2\,\si{\ohm}$}, i>^=$I_2$] (5.6,0);
    \draw (0,0) -- (5.6,0);
    \draw (4.0,0) node[ground] {} node[below right] {$0\,\si{\volt}$};
\end{circuitikz}
\end{center}

\textbf{Resolución paso a paso}:
\begin{enumerate}
    \item \textbf{Resolución por el método de nudos}: Referenciando el nudo inferior a tierra ($0\,\si{\volt}$), aplicamos LCK en el nudo principal superior $V_A$ ($\mathbf{I}_1 = \mathbf{I}_2 + \mathbf{I}_3 + \mathbf{I}_4$):
    \begin{equation*}
        4 = \frac{\mathbf{V}_A}{2} + \frac{\mathbf{V}_A}{j 1} + \frac{\mathbf{V}_A}{4 - j 2} = \frac{\mathbf{V}_A}{2} - j\mathbf{V}_A + \frac{\mathbf{V}_A(4 + j 2)}{20}
    \end{equation*}
    Multiplicando toda la ecuación por $20$:
    \begin{equation*}
        80 = 10\mathbf{V}_A - j 20\mathbf{V}_A + (4 + j 2)\mathbf{V}_A = (14 - j 18)\mathbf{V}_A \implies \mathbf{V}_A = \frac{80}{14 - j 18} = (2{,}154 + j 2{,}769)\,\si{\volt} = 3{,}508\angle 52{,}12^\circ\,\si{\volt}
    \end{equation*}
    Pasando a radianes ($52{,}12^\circ \approx 0{,}29\pi\,\si{\radian}$):
    \begin{equation*}
        \boxed{v_A(t) = 3{,}51\cos(\omega t + 0{,}29\pi)\,\si{\volt}}
    \end{equation*}
    \item \textbf{Intensidades de cada rama y tensiones $\mathbf{V}_B$ y $\mathbf{V}_{R1}$}:
    \begin{align*}
        \mathbf{I}_1 &= 4\angle 0^\circ\,\si{\ampere} &\implies i_1(t) &= 4\cos(\omega t)\,\si{\ampere} \\
        \mathbf{I}_2 &= \frac{\mathbf{V}_A}{2\angle 0^\circ} = 1{,}754\angle 52{,}12^\circ\,\si{\ampere} &\implies i_2(t) &= 1{,}75\cos(\omega t + 0{,}29\pi)\,\si{\ampere} \\
        \mathbf{I}_3 &= \frac{\mathbf{V}_A}{1\angle 90^\circ} = 3{,}51\angle -37{,}88^\circ\,\si{\ampere} &\implies i_3(t) &= 3{,}51\cos(\omega t - 0{,}21\pi)\,\si{\ampere} \\
        \mathbf{I}_4 &= \frac{\mathbf{V}_A}{4 - j 2} = \frac{3{,}508\angle 52{,}12^\circ}{4{,}472\angle -26{,}56^\circ} = 0{,}784\angle 78{,}68^\circ\,\si{\ampere} &\implies i_4(t) &= 0{,}78\cos(\omega t + 0{,}437\pi)\,\si{\ampere}
    \end{align*}
    \begin{align*}
        \mathbf{V}_B &= \mathbf{I}_4\cdot\mathbf{Z}_C = (0{,}784\angle 78{,}68^\circ)(2\angle -90^\circ) = 1{,}57\angle -11{,}32^\circ\,\si{\volt} \implies v_B(t) = 1{,}57\cos(\omega t - 0{,}063\pi)\,\si{\volt} \\
        \mathbf{V}_{R1} &= \mathbf{V}_A - \mathbf{V}_B = (0{,}61 + j 3{,}08)\,\si{\volt} = 3{,}14\angle 78{,}68^\circ\,\si{\volt} \implies v_{R1}(t) = 3{,}14\cos(\omega t + 0{,}437\pi)\,\si{\volt}
    \end{align*}
    \item \textbf{Potencias medias y comprobación del balance energético}:
    \begin{align*}
        P_{\text{fuente,media}} &= \frac{4 \cdot 3{,}51}{2}\cos(0{,}29\pi) = 7{,}02\cdot 0{,}613 = \boxed{4{,}30\,\si{\watt}} \\
        P_{R2,\text{media}} &= \frac{1{,}754 \cdot 3{,}508}{2}\cos(0) = \boxed{3{,}08\,\si{\watt}}, \qquad P_{R1,\text{media}} = \frac{0{,}784 \cdot 3{,}14}{2}\cos(0) = \boxed{1{,}22\,\si{\watt}} \\
        P_{L,\text{media}} &= \frac{3{,}51\cdot 3{,}51}{2}\cos\left(\frac{\pi}{2}\right) = \boxed{0\,\si{\watt}}, \qquad P_{C,\text{media}} = \frac{0{,}784\cdot 1{,}57}{2}\cos\left(\frac{\pi}{2}\right) = \boxed{0\,\si{\watt}}
    \end{align*}
    Se comprueba exactamente que $P_{\text{fuente,media}} = P_{R1,\text{media}} + P_{R2,\text{media}} = 1{,}22 + 3{,}08 = 4{,}30\,\si{\watt}$.
\end{enumerate}
\end{ejercicio}

\section{Formulario y Resumen General del Tema 3 (AC)}

\begin{ley}[Formulario de Referencia Rápida de Corriente Alterna]
\begin{itemize}
    \item \textbf{Señal alterna y fasor}: $v(t) = V_m\cos(\omega t + \phi) \longleftrightarrow \mathbf{V} = V_m\angle\phi = V_m e^{j\phi} = V_m\cos\phi + j V_m\sin\phi$, con $\omega = \frac{2\pi}{T} = 2\pi f$.
    \item \textbf{Impedancias ($\mathbf{Z} = \mathbf{V}/\mathbf{I}$) y Admitancias ($\mathbf{Y} = 1/\mathbf{Z}$)}:
    \begin{equation*}
        \mathbf{Z}_R = R\angle 0^\circ \;\left(\mathbf{Y}_R = \frac{1}{R}\right), \quad \mathbf{Z}_L = j\omega L = \omega L\angle 90^\circ \;\left(\mathbf{Y}_L = \frac{1}{j\omega L}\right), \quad \mathbf{Z}_C = \frac{-j}{\omega C} = \frac{1}{\omega C}\angle -90^\circ \;\left(\mathbf{Y}_C = j\omega C\right)
    \end{equation*}
    \item \textbf{Función de transferencia y frecuencia de corte}:
    \begin{equation*}
        \mathbf{T}(\omega) = \frac{\mathbf{V}_o(\omega)}{\mathbf{V}_i(\omega)}, \quad |\mathbf{T}(\omega_0)| = \frac{|\mathbf{T}|_{\max}}{\sqrt{2}} \approx 0{,}707\,|\mathbf{T}|_{\max} \;(-3\,\si{\decibel}), \quad |\mathbf{T}|_{\si{\decibel}} = 20\log_{10}|\mathbf{T}(\omega)|
    \end{equation*}
    \item \textbf{Potencia en CA (en el dominio del tiempo)}:
    \begin{equation*}
        p(t) = \frac{V_m I_m}{2}\big[\cos(2\omega t + \theta_v + \theta_i) + \cos(\theta_v - \theta_i)\big], \qquad P_{\text{media}} = \frac{V_m I_m}{2}\cos(\theta_v - \theta_i)
    \end{equation*}
\end{itemize}
\end{ley}
