\section{Señales Sinusoidales en Régimen Permanente y Fasores}

En corriente alterna (AC), las fuentes generan tensiones y corrientes que varían sinusoidalmente en el tiempo:
\begin{equation}
    v(t) = V_m \cos(\omega t + \varphi) \quad [\si{\volt}]
\end{equation}
donde $V_m$ es la amplitud pico, $\omega = 2\pi f = 2\pi/T$ la frecuencia angular y $\varphi$ la fase inicial.

\subsection{Valor Eficaz (RMS) y Fasores}
El valor eficaz mide la equivalencia térmica con corriente continua:
\begin{equation}
    V_{rms} = \frac{V_m}{\sqrt{2}} \approx 0{,}7071 \cdot V_m
\end{equation}
El fasor asociado a la sinusoide contiene la magnitud y fase en el plano complejo:
\begin{equation}
    \mathbf{V} = V_{rms} e^{j\varphi} = V_{rms} \angle \varphi
\end{equation}

\subsection{Impedancia Compleja de los Elementos Básicos}
\begin{definicion}[Impedancia Compleja $\mathbf{Z} = R + jX$]
\begin{itemize}[leftmargin=*]
    \item \textbf{Resistencia Pura ($R$):} $\mathbf{Z}_R = R \angle 0^\circ = R$ (Tensión y corriente están \textbf{en fase}).
    \item \textbf{Bobina Pura ($L$):} $\mathbf{Z}_L = j\omega L = \omega L \angle 90^\circ$ (\textbf{La tensión adelanta $90^\circ$ a la corriente}).
    \item \textbf{Condensador Puro ($C$):} $\mathbf{Z}_C = \frac{1}{j\omega C} = -\frac{j}{\omega C} = \frac{1}{\omega C} \angle -90^\circ$ (\textbf{La corriente adelanta $90^\circ$ a la tensión}).
\end{itemize}
\end{definicion}

\section{Potencia en Corriente Alterna y Compensación Reactiva}

\subsection{El Triángulo de Potencias}
\begin{definicion}[Clasificación de Potencias en AC]
\begin{itemize}[leftmargin=*]
    \item \textbf{Potencia Activa ($P$):} Trabajo útil y pérdidas Joule: $P = V_{rms} I_{rms} \cos \theta \quad [\si{\watt}]$.
    \item \textbf{Potencia Reactiva ($Q$):} Potencia fluctuante oscilante: $Q = V_{rms} I_{rms} \sin \theta \quad [\si{\VAR}]$.
    \item \textbf{Potencia Compleja ($\mathbf{S}$):} $\mathbf{S} = P + jQ = S \angle \theta \quad [\si{\VA}]$.
    \item \textbf{Potencia Aparente ($S$):} $S = |\mathbf{S}| = \sqrt{P^2 + Q^2} = V_{rms} I_{rms} \quad [\si{\VA}]$.
    \item \textbf{Factor de Potencia ($\text{fp}$):} $\text{fp} = \cos \theta = P/S$.
\end{itemize}
\end{definicion}

\begin{tikzpicture}[scale=1.1]

\section{Filtros Pasivos RC y Diagramas de Bode}

\subsection{Filtro RC Pasa-Bajos}
La señal de salida se toma en bornes del condensador:
\begin{equation}
    H(j\omega) = \frac{\mathbf{V}_o}{\mathbf{V}_i} = \frac{\frac{1}{j\omega C}}{R + \frac{1}{j\omega C}} = \frac{1}{1 + j\omega RC} = \frac{1}{1 + j(\omega / \omega_c)}, \qquad \omega_c = \frac{1}{RC}
\end{equation}

\subsection{Filtro RC Pasa-Altos}
La salida se toma en bornes de la resistencia:
\begin{equation}
    H(j\omega) = \frac{\mathbf{V}_o}{\mathbf{V}_i} = \frac{R}{R + \frac{1}{j\omega C}} = \frac{j\omega RC}{1 + j\omega RC} = \frac{j(\omega / \omega_c)}{1 + j(\omega / \omega_c)}
\end{equation}

\section{Ejercicios Resueltos de Exámenes de Circuitos AC}

\begin{ejercicio}[Análisis Fasorial Completo de Circuito RLC --- Examen UGR]
Una fuente sinusoidal de tensión $v_s(t) = 100 \sqrt{2} \cos(1000 t)\ \si{\volt}$ alimenta una asociación serie formada por $R = 30\ \si{\ohm}$, $L = 80\ \si{\milli\henry}$ y $C = 25\ \si{\micro\farad}$.
Hallar la impedancia $\mathbf{Z}_{eq}$, la corriente temporal $i(t)$, las potencias $P, Q, S$ y el factor de potencia.
\tcbline
\textbf{Esquema Circuital en el Dominio Fasorial:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=1.0]
    \draw (0,0) to[sV, v=$\mathbf{V}_s{=}100\angle 0^\circ\si{\volt}$] (0,2)
          to[R, l=$R{=}30\si{\ohm}$] (2.5,2)
          to[L, l=$j\omega L{=}j80\si{\ohm}$] (5,2)
          to[C, l=$-j\frac{1}{\omega C}{=}-j40\si{\ohm}$] (5,0)
          -- (0,0);
    \draw[->, >=stealth, very thick, fftAccent] (0.3,2.3) -- (1.5,2.3) node[midway, above] {$\mathbf{I}$};
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
1) Con $\omega = 1000\ \si{\radian\per\second}$:
\begin{align}
    \mathbf{Z}_R &= 30\ \si{\ohm} \\
    \mathbf{Z}_L &= j(1000 \times 0{,}080) = j80\ \si{\ohm} \\
    \mathbf{Z}_C &= -j\frac{1}{1000 \times (25 \times 10^{-6})} = -j40\ \si{\ohm}
\end{align}
\begin{equation}
    \mathbf{Z}_{eq} = 30 + j(80 - 40) = 30 + j40\ \si{\ohm} = 50 \angle 53{,}13^\circ\ \si{\ohm}
\end{equation}

2) Corriente fasorial y expresión temporal:
\begin{equation}
    \mathbf{I}_{rms} = \frac{\mathbf{V}_s}{\mathbf{Z}_{eq}} = \frac{100 \angle 0^\circ}{50 \angle 53{,}13^\circ} = 2 \angle -53{,}13^\circ\ \si{\ampere}
\end{equation}
\begin{equation}
    i(t) = 2\sqrt{2} \cos(1000 t - 53{,}13^\circ)\ \si{\ampere}
\end{equation}

3) Potencias y Factor de Potencia:
\begin{align}
    P &= V_{rms} I_{rms} \cos(\theta) = 100 \times 2 \times \cos(53{,}13^\circ) = 200 \times 0{,}60 = 120\ \si{\watt} \\
    Q &= V_{rms} I_{rms} \sin(\theta) = 100 \times 2 \times \sin(53{,}13^\circ) = 200 \times 0{,}80 = 160\ \si{\VAR} \\
    S &= V_{rms} I_{rms} = 200\ \si{\VA} \\
    \text{fp} &= \cos(53{,}13^\circ) = 0{,}60\quad (\text{en retraso / inductivo})
\end{align}
\end{ejercicio}

\begin{ejercicio}[Diseño y Respuesta en Frecuencia de un Filtro RC Pasa-Bajos]
Diseñar un filtro RC con $C = 10\ \si{\nano\farad}$ para frecuencia de corte $f_c = 10\ \si{\kilo\hertz}$ y calcular la salida para $V_i = 5\ \si{\volt}$ a $f_1 = 100\ \si{\kilo\hertz}$.
\tcbline
\textbf{Solución Paso a Paso:}\\[4pt]
a) Resistencia requerida:
\begin{equation}
    R = \frac{1}{2\pi f_c C} = \frac{1}{2\pi \times 10^4 \times (10 \times 10^{-9})} \approx 1591{,}55\ \si{\ohm} \approx 1{,}59\ \si{\kilo\ohm}
\end{equation}
b) Para $f_1 = 10 f_c$ ($\omega / \omega_c = 10$):
\begin{equation}
    |H| = \frac{1}{\sqrt{1 + 10^2}} = \frac{1}{\sqrt{101}} \approx 0{,}0995 \implies V_o = 0{,}0995 \times 5 = 0{,}4975\ \si{\volt} \approx 0{,}50\ \si{\volt}
\end{equation}
\begin{equation}
    \varphi_o = -\arctan(10) \approx -84{,}29^\circ
\end{equation}
\end{ejercicio}
