\section{Física de Semiconductores y la Unión PN}

Los dispositivos electrónicos modernos basan su funcionamiento en las propiedades de conducción controlada de los materiales semiconductores (principalmente el Silicio, Si).

\subsection{Estructura de la Unión PN en Equilibrio}
Al poner en contacto un semiconductor tipo P (rico en huecos mayoritarios) con uno tipo N (rico en electrones libres):
\begin{enumerate}[leftmargin=*]
    \item Los electrones difunden hacia la zona P y los huecos hacia la zona N, recombinándose.
    \item Queda al descubierto una \textbf{Zona de Carga Espacial (ZCE)} con iones fijos inmóviles ($N_A^-$ en el lado P y $N_D^+$ en el lado N).
    \item Se origina un \textbf{campo eléctrico interno} $\vec{E}_{int}$ que frena la difusión y establece una barrera de potencial de contacto $V_{bi} \approx 0{,}6 - 0{,}7\ \si{\volt}$ para Silicio.
\end{enumerate}

\section{El Diodo de Unión PN: Polarización y Modelos Circuitales}

\subsection{Curva Característica $I-V$ y Modelos Lineales}
La relación corriente-tensión real viene dada por la ecuación de Shockley:
\begin{equation}
    I_D = I_S \left( e^{\frac{V_D}{n V_T}} - 1 \right)
\end{equation}

\section{Ejercicios Resueltos de Exámenes de Diodos y MOSFET}

\begin{ejercicio}[Circuito con Diodos y Análisis de Estados --- Examen Parcial 2 FFT]
Determinar el estado de conducción (ON/OFF), la corriente y las caídas de tensión en el circuito empleando $V_\gamma = 0{,}7\ \si{\volt}$.
\tcbline
\textbf{Esquema Circuital Completo:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    % Fuente de 10V
    \draw (0,0) to[V, v=$10\si{\volt}$] (0,2.5) -- (2,2.5) node[above] {$A$};
    % Rama 1: D1 y R1
    \draw (2,2.5) to[D, l=$D_1$] (4,2.5)
          to[R, l=$R_1{=}1\si{\kilo\ohm}$, i=$I_{D1}$] (4,0) -- (0,0);
    % Rama 2: D2, V2 y R2
    \draw (2,2.5) -- (2,4) to[D, l=$D_2$] (4,4)
          to[V, v=$V_2{=}4\si{\volt}$] (6,4)
          to[R, l=$R_2{=}2\si{\kilo\ohm}$, i=$I_{D2}$] (6,0) -- (4,0);
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
El potencial en el nudo $A$ común a ambos ánodos es $V_A = 10\ \si{\volt}$.\\
\textbf{Hipótesis:} Ambos diodos conducen ($D_1$ ON y $D_2$ ON, sustituidos por baterías de $V_\gamma = 0{,}7\ \si{\volt}$).
\begin{align}
    V_{K1} &= V_A - V_\gamma = 10 - 0{,}7 = 9{,}3\ \si{\volt} \implies I_{D1} = \frac{9{,}3\ \si{\volt}}{1\ \si{\kilo\ohm}} = 9{,}3\ \si{\milli\ampere} > 0 \\
    V_{K2} &= V_A - V_\gamma = 10 - 0{,}7 = 9{,}3\ \si{\volt} \implies I_{D2} = \frac{9{,}3 - 4}{2\ \si{\kilo\ohm}} = 2{,}65\ \si{\milli\ampere} > 0
\end{align}
Ambas corrientes son positivas $\implies$ Hipótesis confirmada.\\
Corriente total: $I_{total} = I_{D1} + I_{D2} = 9{,}3 + 2{,}65 = 11{,}95\ \si{\milli\ampere}$.
\end{ejercicio}

\begin{ejercicio}[Punto de Trabajo Q de Transistor NMOS --- Examen UGR]
Determinar el punto de trabajo $Q(V_{GS}, I_D, V_{DS})$ en el circuito de polarización del transistor NMOS con $V_{th} = 1{,}5\ \si{\volt}$, $k_n = 1\ \si{\milli\ampere\per\volt\squared}$, $\lambda = 0$, $V_{DD} = 15\ \si{\volt}$, $R_1 = 10\ \si{\mega\ohm}$, $R_2 = 5\ \si{\mega\ohm}$, $R_D = 3\ \si{\kilo\ohm}$ y $R_S = 1\ \si{\kilo\ohm}$.
\tcbline
\textbf{Esquema Circuital de Polarización:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=1.0]
    % Fuente de alimentación VDD
    \draw (0,4) node[above] {$V_{DD} = 15\si{\volt}$} -- (0,3.6);
    \draw (3,4) node[above] {$V_{DD} = 15\si{\volt}$} -- (3,3.6);

    % Divisor de puerta
    \draw (0,3.6) to[R, l=$R_1{=}10\si{\mega\ohm}$] (0,2)
          to[R, l=$R_2{=}5\si{\mega\ohm}$] (0,0) node[ground] {};
    % Conexión a la puerta
    \draw (0,2) -- (1.5,2);

    % Transistor NMOS
    \draw (2.0,2) node[nmos] (nmos) {};
    \draw (1.5,2) -- (nmos.gate);

    % Drenador
    \draw (3,3.6) to[R, l=$R_D{=}3\si{\kilo\ohm}$] (3,2.5) -- (nmos.drain);

    % Fuente
    \draw (nmos.source) -- (2.0,1.2)
          to[R, l=$R_S{=}1\si{\kilo\ohm}$] (2.0,0) node[ground] {};
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
1) Tensión de Puerta ($I_G = 0$):
\begin{equation}
    V_G = V_{DD} \frac{R_2}{R_1 + R_2} = 15 \times \frac{5}{15} = 5\ \si{\volt}
\end{equation}
2) Malla Puerta-Fuente ($V_S = I_D R_S = 1 \cdot I_D$):
\begin{equation}
    V_{GS} = 5 - I_D
\end{equation}
3) Hipótesis de Saturación:
\begin{equation}
    I_D = \frac{1}{2} k_n (V_{GS} - V_{th})^2 = 0{,}5 (3{,}5 - I_D)^2 \implies I_D^2 - 9 I_D + 12{,}25 = 0
\end{equation}
Raíz físicamente admisible:
\begin{equation}
    I_D = \frac{9 - \sqrt{32}}{2} \approx 1{,}67\ \si{\milli\ampere} \implies V_{GS} = 5 - 1{,}67 = 3{,}33\ \si{\volt} > V_{th}
\end{equation}
4) Comprobación de Saturación:
\begin{equation}
    V_D = 15 - 3 \times 1{,}67 = 9{,}99\ \si{\volt}, \quad V_S = 1{,}67\ \si{\volt} \implies V_{DS} = 8{,}32\ \si{\volt}
\end{equation}
Como $V_{DS} = 8{,}32\ \si{\volt} \ge V_{GS} - V_{th} = 1{,}83\ \si{\volt}$, queda verificada la saturación.\\
\textbf{Punto Q:} $I_D = 1{,}67\ \si{\milli\ampere}, \quad V_{GS} = 3{,}33\ \si{\volt}, \quad V_{DS} = 8{,}32\ \si{\volt}$.
\end{ejercicio}
