\subsection{Niveles Lógicos y Márgenes de Ruido}
\begin{definicion}[Niveles de Tensión y Márgenes de Ruido]
Para garantizar la fiabilidad del bit transmitido entre puertas digitales:
\begin{align}
    NM_H &= V_{OH} - V_{IH} \quad (\text{Margen de Ruido a Nivel Alto}) \\
    NM_L &= V_{IL} - V_{OL} \quad (\text{Margen de Ruido a Nivel Bajo})
\end{align}
\end{definicion}

\section{Síntesis de Puertas Lógicas CMOS Combinacionales}

\begin{ley}[Reglas de Dualidad de De Morgan en CMOS]
\begin{itemize}[leftmargin=*]
    \item \textbf{En la red PDN (NMOS):} AND $\implies$ \textbf{Serie}; OR $\implies$ \textbf{Paralelo}.
    \item \textbf{En la red PUN (PMOS):} AND $\implies$ \textbf{Paralelo}; OR $\implies$ \textbf{Serie}.
\end{itemize}
\end{ley}

\section{Ejercicios Resueltos de Exámenes de Electrónica Digital}

\begin{ejercicio}[Síntesis de Función Compleja CMOS --- Examen Enero 2020 FFT]
Diseñar a nivel de transistores la puerta estática complementaria CMOS para:
\begin{equation}
    Y = \overline{(A + B) \cdot C + D}
\end{equation}
\tcbline
\textbf{Esquema Circuital Completo en Tecnología CMOS (8 Transistores):}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    % VDD arriba
    \draw (0,5.8) node[above] {$V_{DD}$} -- (0,5.4);

    % RED PUN: PD en serie con [PC paralelo (PA serie PB)]
    \draw (0,5.0) node[pmos] (pd) {};
    \node[left] at (pd.gate) {$D$};

    \draw (pd.drain) -- (0,4.2);
    % Paralelo: PC a la izquierda, PA-PB a la derecha
    \draw (0,4.2) -- (-1.6,4.2) node[pmos] (pc) {};
    \node[left] at (pc.gate) {$C$};
    \draw (0,4.2) -- (1.6,4.2) node[pmos] (pa) {};
    \node[right] at (pa.gate) {$A$};
    \draw (pa.drain) node[pmos] (pb) {};
    \node[right] at (pb.gate) {$B$};

    \draw (pc.drain) -- (-1.6,2.2) -- (1.6,2.2) -- (pb.drain);
    \draw (0,2.2) -- (0,1.8);
    \draw (0,2.0) to[short, -o] (2.8,2.0) node[right, font=\bfseries] {$Y = \overline{(A+B)C + D}$};

    % RED PDN: [(NA paralelo NB) serie NC] paralelo ND
    \draw (0,1.8) -- (-1.6,1.8);
    % Rama ND
    \draw (0,1.8) -- (1.6,1.8) node[nmos] (nd) {};
    \node[right] at (nd.gate) {$D$};

    % Rama NC serie (NA par NB)
    \draw (-1.6,1.4) node[nmos] (nc) {};
    \node[left] at (nc.gate) {$C$};
    \draw (nc.source) -- (-1.6,0.6);
    \draw (-1.6,0.6) -- (-2.4,0.6) node[nmos] (na) {};
    \node[left] at (na.gate) {$A$};
    \draw (-1.6,0.6) -- (-0.8,0.6) node[nmos] (nb) {};
    \node[right] at (nb.gate) {$B$};

    \draw (na.source) -- (-2.4,-0.4) -- (-0.8,-0.4) -- (nb.source);
    \draw (-1.6,-0.4) -- (1.6,-0.4) -- (nd.source);
    \draw (0,-0.4) -- (0,-0.8) node[ground] {};
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[2pt]
- \textbf{Red PDN (NMOS):} $N_A \parallel N_B$ en serie con $N_C$, y todo ello en paralelo con $N_D$ hacia masa.\\[2pt]
- \textbf{Red PUN (PMOS):} Transistor $P_D$ en serie con el bloque $[P_C \parallel (P_A \text{ serie } P_B)]$ hacia $V_{DD}$.\\[2pt]
- Total: 4 NMOS $+ 4$ PMOS = \textbf{8 transistores}.
\end{ejercicio}
