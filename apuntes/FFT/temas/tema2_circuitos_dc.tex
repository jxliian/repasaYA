\section{Elementos del Circuito y Convención de Signos}

Un circuito eléctrico es una interconexión de componentes con terminales a través de los cuales fluye carga eléctrica. En corriente continua (DC), las tensiones y corrientes no varían con el tiempo en régimen permanente.

\subsection{Convención de Signos y Fuentes de Energía}
\begin{definicion}[Convención de Signos Pasiva y Activa]
\begin{itemize}[leftmargin=*]
    \item \textbf{Criterio Pasivo (Receptor):} La corriente $I$ entra por el terminal positivo ($+$). La potencia absorbida es $P = V \cdot I$. Si $P > 0$, el elemento disipa o almacena energía.
    \item \textbf{Criterio Activo (Generador):} La corriente $I$ sale por el terminal positivo ($+$). La potencia entregada es $P_{entregada} = V \cdot I$.
\end{itemize}
\end{definicion}

\section{Leyes de Kirchhoff y Divisores de Tensión y Corriente}

\begin{ley}[Leyes Fundamentales de Kirchhoff]
\begin{itemize}[leftmargin=*]
    \item \textbf{Primera Ley de Kirchhoff o Ley de Corrientes (LCK):} En cualquier nudo, la suma de corrientes entrantes es igual a la suma de salientes: $\sum I_{entrantes} = \sum I_{salientes}$.
    \item \textbf{Segunda Ley de Kirchhoff o Ley de Tensiones (LTK):} A lo largo de cualquier lazo cerrado, la suma de caídas y subidas de potencial es nula: $\sum V_k = 0$.
\end{itemize}
\end{ley}

\section{Teoremas de Thévenin, Norton y Máxima Potencia}

\begin{ley}[Teoremas de Thévenin y Norton]
Cualquier red lineal de dos terminales $A$ y $B$ puede ser sustituida formalmente por un circuito equivalente de Thévenin (fuente de tensión $V_{th}$ en serie con $R_{th}$) o de Norton (fuente de corriente $I_N = V_{th}/R_{th}$ en paralelo con $R_N = R_{th}$).
\end{ley}

\begin{center}
\begin{tikzpicture}[scale=0.95]
    % Equivalente Thévenin
    \node[font=\bfseries] at (-3,2.2) {Equivalente de Thévenin};
    \draw (-4.5,0) to[V, v=$V_{th}$] (-4.5,1.5)
          to[R, l=$R_{th}$] (-2,1.5)
          to[short, -o] (-1.2,1.5) node[right] {$A$};
    \draw (-4.5,0) to[short, -o] (-1.2,0) node[right] {$B$};
    \draw (-1.2,1.5) to[R, l=$R_L$, dashed] (-1.2,0);

    % Equivalente Norton
    \node[font=\bfseries] at (3.5,2.2) {Equivalente de Norton};
    \draw (1.8,0) to[I, i=$I_N$] (1.8,1.5)
          -- (3.2,1.5) to[R, l=$R_N$] (3.2,0) -- (1.8,0);
    \draw (3.2,1.5) to[short, -o] (4.2,1.5) node[right] {$A$};
    \draw (3.2,0) to[short, -o] (4.2,0) node[right] {$B$};
    \draw (4.2,1.5) to[R, l=$R_L$, dashed] (4.2,0);
\end{tikzpicture}
\end{center}

\begin{ley}[Teorema de Máxima Transferencia de Potencia]
La potencia transferida a una carga resistiva $R_L$ conectada a los terminales de Thévenin es máxima cuando la resistencia de carga se iguala exactamente a la resistencia interna de Thévenin:
\begin{equation}
    R_L = R_{th} \implies P_{max} = \frac{V_{th}^2}{4 R_{th}}
\end{equation}
\end{ley}

\begin{center}
\begin{tikzpicture}[scale=0.9]
    \draw[->, thick] (0,0) -- (5,0) node[right] {$R_L$};
    \draw[->, thick] (0,0) -- (0,3) node[above] {$P_L(R_L)$};
    % Curva de potencia
    \draw[very thick, fftAccent, domain=0.05:4.8, samples=80] 
        plot (\x, {2.6 * 4 * 1.5 * \x / ((1.5 + \x)^2)});
    % Líneas de trazo en el máximo

\end{tikzpicture}
\end{center}

    \draw[dashed, gray] (1.5,2.0) node[left, text=black] {$V_s$} -- (6.0,2.0);
    \draw[dashed, gray] (2.5,0) node[below, text=black] {$\tau$} -- (2.5, 1.264);
    \draw[dashed, gray] (1.5, 1.264) node[left, text=black] {$0{,}632 V_s$} -- (2.5, 1.264);
    \filldraw[fftDarkBlue] (2.5, 1.264) circle (2pt);
\end{tikzpicture}
\end{center}

\begin{definicion}[Ecuaciones Fundamentales del Transitorio RC]
\begin{align}
    v_C(t) &= V_s \left(1 - e^{-t/\tau}\right), \quad \text{con } \tau = R C \\
    i_C(t) &= \frac{V_s}{R} e^{-t/\tau}
\end{align}
Para $t \ge 5\tau$, $v_C \approx 0{,}993 V_s \approx V_s$, considerándose alcanzado el régimen permanente de continua (condensador cargado como circuito abierto).
\end{definicion}

\section{Ejercicios Resueltos de Exámenes de Circuitos DC}

\begin{ejercicio}[Cálculo de Equivalente Thévenin y Máxima Potencia --- Examen UGR]
Dado el circuito con bornes $A$ y $B$, compuesto por $V_1 = 20\ \si{\volt}$, $I_1 = 2\ \si{\ampere}$, $R_1 = 10\ \si{\ohm}$, $R_2 = 40\ \si{\ohm}$ y $R_3 = 12\ \si{\ohm}$:
\begin{enumerate}[label=\alph*)]
    \item Calcular el equivalente de Thévenin entre los terminales $A$ y $B$.
    \item Determinar la resistencia $R_L$ para absorber la máxima potencia y el valor de dicha potencia.
\end{enumerate}
\tcbline
\textbf{Paso 1: Esquema del Circuito Original Completo:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    % Fuente de tensión y R1
    \draw (0,0) to[V, v=$V_1{=}20\si{\volt}$] (0,2.5)
          to[R, l=$R_1{=}10\si{\ohm}$] (2.5,2.5);
    % Fuente de corriente en paralelo
    \draw (2.5,0) to[I, i=$I_1{=}2\si{\ampere}$] (2.5,2.5);
    % Resistencia R2 en paralelo
    \draw (4.5,2.5) to[R, l=$R_2{=}40\si{\ohm}$] (4.5,0);
    \draw (2.5,2.5) -- (4.5,2.5);
    % Resistencia R3 hacia el terminal A
    \draw (4.5,2.5) to[R, l=$R_3{=}12\si{\ohm}$] (7,2.5) to[short, -o] (7.5,2.5) node[right] {$A$};
    % Conductor de retorno hacia B
    \draw (0,0) -- (7,0) to[short, -o] (7.5,0) node[right] {$B$};
\end{tikzpicture}
\end{center}

\textbf{Paso 2: Cálculo de $V_{th}$ en Circuito Abierto:}\\[2pt]
En circuito abierto entre $A$ y $B$, por $R_3$ no circula corriente ($I_{R3} = 0$), luego $V_A = V_{nudo}$ (potencial común entre $R_1, I_1, R_2$).
Tomando el borne $B$ como referencia ($V_B = 0$):
\begin{equation}
    \frac{V_{th} - V_1}{R_1} - I_1 + \frac{V_{th}}{R_2} = 0 \implies \frac{V_{th} - 20}{10} - 2 + \frac{V_{th}}{40} = 0
\end{equation}
Multiplicamos toda la ecuación por $40$:
\begin{equation}
    4(V_{th} - 20) - 80 + V_{th} = 0 \implies 5 V_{th} - 160 = 0 \implies V_{th} = 32\ \si{\volt}
\end{equation}

\textbf{Paso 3: Cálculo de $R_{th}$ con Fuentes Desactivadas:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.9]
    % Fuente V1 en cortocircuito
    \draw (0,0) -- (0,2) to[R, l=$R_1{=}10\si{\ohm}$] (2,2);
    % Fuente I1 en circuito abierto (omitida)
    % R2 en paralelo
    \draw (2,2) to[R, l=$R_2{=}40\si{\ohm}$] (2,0) -- (0,0);
    % R3 hacia bornes
    \draw (2,2) to[R, l=$R_3{=}12\si{\ohm}$] (4.5,2) to[short, -o] (5,2) node[right] {$A$};
    \draw (2,0) -- (4.5,0) to[short, -o] (5,0) node[right] {$B$};
    \node[right] at (5.2,1) {$\Longrightarrow R_{th} = R_3 + (R_1 \parallel R_2) = 12 + 8 = 20\ \si{\ohm}$};
\end{tikzpicture}
\end{center}
\begin{equation}
    R_{th} = R_3 + \frac{R_1 R_2}{R_1 + R_2} = 12 + \frac{10 \times 40}{50} = 12 + 8 = 20\ \si{\ohm}
\end{equation}

\textbf{Paso 4: Equivalente de Thévenin con Carga Óptima:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    \draw (0,0) to[V, v=$V_{th}{=}32\si{\volt}$] (0,2)
          to[R, l=$R_{th}{=}20\si{\ohm}$] (2.5,2)
          to[short, -o] (3.2,2) node[above] {$A$}
          to[R, l=$R_L{=}20\si{\ohm}$, i=$I_L$] (3.2,0)
          to[short, o-] (2.5,0) -- (0,0);
    \node[below] at (3.2,0) {$B$};
\end{tikzpicture}
\end{center}
Por el teorema de máxima transferencia de potencia:
\begin{equation}
    R_L = R_{th} = 20\ \si{\ohm} \implies P_{max} = \frac{V_{th}^2}{4 R_{th}} = \frac{32^2}{4 \times 20} = \frac{1024}{80} = 12{,}8\ \si{\watt}
\end{equation}
\end{ejercicio}

\begin{ejercicio}[Régimen Transitorio de Carga RC --- Examen Convocatoria Ordinaria]
En un circuito de carga RC, $V_s = 12\ \si{\volt}$, $R = 100\ \si{\kilo\ohm}$ y $C = 50\ \si{\micro\farad}$ con $v_C(0^-) = 0$. En $t = 0$ se cierra el interruptor.
Hallar $\tau$, las expresiones de $v_C(t)$ e $i(t)$, los valores en $t = 5\ \si{\second}$ y el tiempo para alcanzar el $90\%$ de la carga final.
\tcbline
\textbf{Solución Paso a Paso:}\\[4pt]
1) Constante de tiempo:
\begin{equation}
    \tau = R \cdot C = (100 \times 10^3\ \si{\ohm}) \times (50 \times 10^{-6}\ \si{\farad}) = 5{,}0\ \si{\second}
\end{equation}
2) Ecuaciones temporales para $t \ge 0$:
\begin{equation}
    v_C(t) = 12 \left(1 - e^{-t/5}\right)\ \si{\volt}, \qquad i(t) = \frac{12}{100 \times 10^3} e^{-t/5} = 0{,}12 \cdot e^{-t/5}\ \si{\milli\ampere}
\end{equation}
3) En $t = 5\ \si{\second} = \tau$:
\begin{equation}
    v_C(5) = 12 (1 - e^{-1}) \approx 7{,}59\ \si{\volt}, \qquad i(5) = 0{,}12 \cdot e^{-1} \approx 44{,}1\ \si{\micro\ampere}
\end{equation}
4) Para $v_C(t_1) = 0{,}90 \times 12 = 10{,}8\ \si{\volt}$:
\begin{equation}
    1 - e^{-t_1 / 5} = 0{,}90 \implies t_1 = -5 \ln(0{,}10) \approx 11{,}51\ \si{\second}
\end{equation}
\end{ejercicio}
