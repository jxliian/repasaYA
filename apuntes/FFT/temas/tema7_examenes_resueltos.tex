\begin{ejercicio}[Parcial 1 --- Problema 1: Ley de Ampère, Campo Magnético y Fuerza entre Tres Hilos Paralelos]
\begin{enumerate}[label=\arabic*)]
    \item Calcular utilizando la \textbf{Ley de Ampère} la expresión del campo magnético $\vec{B}$ que crea un cable conductor rectilíneo indefinido por el que circula una intensidad $I$ a una distancia radial $r$ (o $d$).
    \item Se tienen tres cables conductores rectilíneos y paralelos de longitud $L = 50\ \si{\centi\meter} = 0{,}5\ \si{\meter}$ (mucho mayor que la distancia que los separa), situados en el vacío ($\mu_0 = 4\pi \times 10^{-7}\ \si{\tesla\meter\per\ampere}$), por los cuales circulan corrientes continuas en el mismo sentido:
    \begin{equation}
        I_1 = 2\ \si{\milli\ampere}, \qquad I_2 = 4\ \si{\milli\ampere}, \qquad I_3 = 6\ \si{\milli\ampere}
    \end{equation}
    \begin{enumerate}[label=\alph*)]
        \item Calcular los campos magnéticos que producen los cables sobre cada uno de los otros dos indicando su módulo, dirección y sentido.
        \item Calcular las fuerzas magnéticas netas que se producen debido a estos campos sobre cada uno de los tres cables, indicando su dirección y sentido, y verificar la Tercera Ley de Newton.
    \end{enumerate}
\end{enumerate}
\tcbline
\textbf{1) Deducción del Campo Magnético de un Hilo Rectilíneo mediante la Ley de Ampère:}\\[4pt]
La \textbf{Ley de Ampère} establece que la circulación del vector campo magnético $\vec{B}$ a lo largo de cualquier línea cerrada amperiana $C$ es proporcional a la intensidad neta de corriente $I_{\text{enc}}$ que atraviesa la superficie delimitada por dicha curva:
\begin{equation}
    \oint_C \vec{B} \cdot d\vec{\ell} = \mu_0 I_{\text{enc}}
\end{equation}

\begin{center}
\begin{tikzpicture}[scale=1.0]
    % Circunferencia amperiana en perspectiva (elipse)
    \draw[thick, dashed, fftPrimary] (0,0) ellipse (2.0cm and 0.65cm);
    % Hilo conductor vertical
    \draw[very thick, fftDarkBlue] (0,-1.6) -- (0,0);
    \draw[very thick, fftDarkBlue, ->, >=stealth] (0,0) -- (0,1.8) node[left] {$I$};
    % Radio r y vectores B y dl
    \draw[->, >=stealth, gray] (0,0) -- (2.0,0) node[midway, above, text=black] {$r$};
    \draw[->, >=stealth, very thick, fftAccent] (2.0,0) -- (2.0,0.95) node[right] {$\vec{B}$};
    \draw[->, >=stealth, thick, fftGreen] (1.41,-0.46) -- (1.85,-0.15) node[below right] {$d\vec{\ell}$};
    \node[below, font=\footnotesize\sffamily] at (0,-1.7) {Curva cerrada amperiana circular de radio $r$};
\end{tikzpicture}
\end{center}

Tomando como camino cerrado $C$ una circunferencia de radio $r$ centrada en el hilo y contenida en el plano perpendicular al conductor, por simetría cilíndrica el campo $\vec{B}$ tiene el mismo módulo en todos los puntos de la circunferencia y es tangente a ella ($\vec{B} \parallel d\vec{\ell} \implies \theta = 0^\circ$, $\cos 0^\circ = 1$):
\begin{equation}
    \oint_C \vec{B} \cdot d\vec{\ell} = \oint_C B\, d\ell \cos(0^\circ) = B \oint_C d\ell = B (2\pi r) = \mu_0 I \implies \boxed{B(r) = \frac{\mu_0 I}{2\pi r}}
\end{equation}

\textbf{Resolución Numérica Paso a Paso (Caso A: $d_{12} = 1\ \si{\centi\meter}$, $d_{23} = 2\ \si{\centi\meter} \implies d_{13} = 3\ \si{\centi\meter}$):}\\[4pt]
Considerando las corrientes salientes del plano ($+\hat{k}$):
\begin{itemize}[leftmargin=*]
    \item \textbf{Sobre el Cable 1 ($x = 0$):} Los cables 2 y 3 se sitúan a su derecha y producen en $H_1$ campos dirigidos verticalmente hacia arriba ($+\hat{j}$):
    \begin{align}
        B_{2 \to 1} &= \frac{\mu_0 I_2}{2\pi d_{12}} = \frac{(4\pi \times 10^{-7}) \cdot (4 \times 10^{-3})}{2\pi \cdot (1 \times 10^{-2})} = 8 \times 10^{-8}\ \si{\tesla} \\
        B_{3 \to 1} &= \frac{\mu_0 I_3}{2\pi d_{13}} = \frac{(4\pi \times 10^{-7}) \cdot (6 \times 10^{-3})}{2\pi \cdot (3 \times 10^{-2})} = 4 \times 10^{-8}\ \si{\tesla} \\
        \vec{B}_1 &= (8 + 4) \times 10^{-8}\,\hat{j} = +12 \times 10^{-8}\ \si{\tesla}\,\hat{j} = 0{,}12\ \si{\micro\tesla}\,\hat{j}
    \end{align}
    Fuerza sobre el cable 1 ($\vec{F}_1 = I_1 \vec{L} \times \vec{B}_1$, hacia la derecha, atracción hacia 2 y 3):
    \begin{equation}
        |\vec{F}_1| = I_1 L B_1 = (2 \times 10^{-3}\ \si{\ampere}) \cdot (0{,}5\ \si{\meter}) \cdot (12 \times 10^{-8}\ \si{\tesla}) = 1{,}2 \times 10^{-10}\ \si{\newton}\ (\text{hacia la derecha})
    \end{equation}

    \item \textbf{Sobre el Cable 2 ($x = 1\ \si{\centi\meter}$):} El cable 1 (a su izquierda) crea un campo hacia abajo ($-\hat{j}$) y el cable 3 (a su derecha) crea un campo hacia arriba ($+\hat{j}$):
    \begin{align}
        B_{1 \to 2} &= \frac{\mu_0 I_1}{2\pi d_{12}} = \frac{(4\pi \times 10^{-7}) \cdot (2 \times 10^{-3})}{2\pi \cdot (1 \times 10^{-2})} = 4 \times 10^{-8}\ \si{\tesla}\ (-\hat{j}) \\
        B_{3 \to 2} &= \frac{\mu_0 I_3}{2\pi d_{23}} = \frac{(4\pi \times 10^{-7}) \cdot (6 \times 10^{-3})}{2\pi \cdot (2 \times 10^{-2})} = 6 \times 10^{-8}\ \si{\tesla}\ (+\hat{j}) \\
        \vec{B}_2 &= (-4 + 6) \times 10^{-8}\,\hat{j} = +2 \times 10^{-8}\ \si{\tesla}\,\hat{j}
    \end{align}
    Fuerza sobre el cable 2:
    \begin{equation}
        |\vec{F}_2| = I_2 L B_2 = (4 \times 10^{-3}\ \si{\ampere}) \cdot (0{,}5\ \si{\meter}) \cdot (2 \times 10^{-8}\ \si{\tesla}) = 0{,}4 \times 10^{-10}\ \si{\newton}\ (\text{hacia la derecha})
    \end{equation}

    \item \textbf{Sobre el Cable 3 ($x = 3\ \si{\centi\meter}$):} Tanto el cable 1 como el cable 2 están a su izquierda, creando campos hacia abajo ($-\hat{j}$):
    \begin{align}
        B_{1 \to 3} &= \frac{\mu_0 I_1}{2\pi d_{13}} = \frac{(4\pi \times 10^{-7}) \cdot (2 \times 10^{-3})}{2\pi \cdot (3 \times 10^{-2})} = 1{,}333 \times 10^{-8}\ \si{\tesla}\ (-\hat{j}) \\
        B_{2 \to 3} &= \frac{\mu_0 I_2}{2\pi d_{23}} = \frac{(4\pi \times 10^{-7}) \cdot (4 \times 10^{-3})}{2\pi \cdot (2 \times 10^{-2})} = 4 \times 10^{-8}\ \si{\tesla}\ (-\hat{j}) \\
        \vec{B}_3 &= -(1{,}333 + 4) \times 10^{-8}\,\hat{j} = -5{,}333 \times 10^{-8}\ \si{\tesla}\,\hat{j}
    \end{align}
    Fuerza sobre el cable 3:
    \begin{equation}
        |\vec{F}_3| = I_3 L B_3 = (6 \times 10^{-3}\ \si{\ampere}) \cdot (0{,}5\ \si{\meter}) \cdot (5{,}333 \times 10^{-8}\ \si{\tesla}) = 1{,}6 \times 10^{-10}\ \si{\newton}\ (\text{hacia la izquierda})
    \end{equation}
\end{itemize}
\textbf{Comprobación de acción y reacción:} $\sum \vec{F} = (+1{,}2 + 0{,}4 - 1{,}6) \times 10^{-10}\ \si{\newton} = 0\ \si{\newton}$.\\[6pt]
\textbf{Variante del Manuscrito Original (Caso B: $d_{12} = 1\ \si{\centi\meter}$, $d_{23} = 1\ \si{\centi\meter} \implies d_{13} = 2\ \si{\centi\meter}$):}
Si ambos pares de hilos adyacentes distan $1\ \si{\centi\meter}$ ($d_{13} = 2\ \si{\centi\meter}$):
\begin{itemize}[leftmargin=*]
    \item En el \textbf{Hilo 1}: $B_{21} = 8 \times 10^{-8}\ \si{\tesla}$, $B_{31} = \frac{2 \times 10^{-7} \cdot 6 \times 10^{-3}}{2 \times 10^{-2}} = 6 \times 10^{-8}\ \si{\tesla} \implies B_{\text{en } 1} = 14 \times 10^{-8}\ \si{\tesla}$ (hacia afuera) y $F_{\text{en } 1} = 0{,}5 \cdot 2 \times 10^{-3} \cdot 14 \times 10^{-8} = 1{,}4 \times 10^{-10}\ \si{\newton}$ (hacia la derecha).
    \item En el \textbf{Hilo 2}: $B_{12} = 4 \times 10^{-8}\ \si{\tesla}$ (hacia adentro), $B_{32} = 12 \times 10^{-8}\ \si{\tesla}$ (hacia afuera) $\implies B_{\text{en } 2} = (12 - 4) \times 10^{-8} = 8 \times 10^{-8}\ \si{\tesla}$ (hacia afuera) y $F_{\text{en } 2} = 0{,}5 \cdot 4 \times 10^{-3} \cdot 8 \times 10^{-8} = 1{,}6 \times 10^{-10}\ \si{\newton}$ (hacia la derecha).
    \item En el \textbf{Hilo 3}: $B_{13} = 2 \times 10^{-8}\ \si{\tesla}$ (hacia adentro), $B_{23} = 8 \times 10^{-8}\ \si{\tesla}$ (hacia adentro) $\implies B_{\text{en } 3} = 10 \times 10^{-8}\ \si{\tesla}$ (hacia adentro) y $F_{\text{en } 3} = 0{,}5 \cdot 6 \times 10^{-3} \cdot 10 \times 10^{-8} = 3{,}0 \times 10^{-10}\ \si{\newton}$ (hacia la izquierda).
\end{itemize}
Nuevamente se verifica el equilibrio interno: $(1{,}4 + 1{,}6 - 3{,}0) \times 10^{-10}\ \si{\newton} = 0\ \si{\newton}$.
\end{ejercicio}

\textbf{Solución Paso a Paso:}\\[4pt]
\textbf{1) Relaciones de tensión inmediatas y supernudo:}
Tomando el nodo inferior como referencia ($V_f = 0\ \si{\volt}$):
\begin{equation}
    V_a = 12\ \si{\volt}, \qquad V_b - V_d = 22\ \si{\volt} \implies V_b = V_d + 22\ \si{\volt}
\end{equation}
\textbf{2) Expresión de las corrientes de rama en función de $V_d$ y $V_e$:}
\begin{align}
    i_5 &= \frac{V_d - V_f}{R_4} = \frac{V_d}{1\ \si{\ohm}} = V_d \\
    i_3 &= i_5 - 1\ \si{\ampere} = V_d - 1 \qquad (\text{aplicando LCK en el nudo } d:\ i_3 + 1 = i_5) \\
    i_1 &= \frac{V_b - V_a}{R_1} = \frac{(V_d + 22) - 12}{2} = \frac{V_d + 10}{2} \\
    i_2 &= \frac{V_b - V_e}{R_2 + R_3} = \frac{V_d + 22 - V_e}{4 + 1} = \frac{V_d + 22 - V_e}{5} \\
    i_4 &= \frac{V_e - V_f}{R_5} = \frac{V_e}{8}
\end{align}
\textbf{3) Sistema de ecuaciones de Kirchhoff en los nudos $b$ y $e$:}
\begin{itemize}[leftmargin=*]
    \item \textbf{En el nudo $b$} ($i_1 + i_2 + i_3 = 0$):
    \begin{equation}
        \frac{V_d + 10}{2} + \frac{V_d + 22 - V_e}{5} + (V_d - 1) = 0 \xrightarrow{\times 10} 5V_d + 50 + 2V_d + 44 - 2V_e + 10V_d - 10 = 0
    \end{equation}
    \begin{equation}
        17 V_d - 2 V_e = -84
    \end{equation}
    \item \textbf{En el nudo $e$} ($i_2 = 1\ \si{\ampere} + i_4$):
    \begin{equation}
        \frac{V_d + 22 - V_e}{5} = 1 + \frac{V_e}{8} \xrightarrow{\times 40} 8V_d + 176 - 8V_e = 40 + 5V_e
    \end{equation}
    \begin{equation}
        8 V_d - 13 V_e = -136
    \end{equation}
\end{itemize}
\textbf{4) Resolución del sistema de tensiones de nudo y corrientes de rama:}
Resolviendo el sistema $2 \times 2$ se obtiene:
\begin{equation}
    \boxed{V_d = -4\ \si{\volt}, \quad V_e = 8\ \si{\volt}, \quad V_b = 18\ \si{\volt}, \quad V_a = 12\ \si{\volt}, \quad V_c = V_e + i_2 R_3 = 10\ \si{\volt}}
\end{equation}
Sustituyendo en las expresiones de las intensidades:
\begin{equation}
    \boxed{i_1 = \frac{-4 + 10}{2} = 3\ \si{\ampere}, \quad i_2 = \frac{10}{5} = 2\ \si{\ampere}, \quad i_3 = -4 - 1 = -5\ \si{\ampere}, \quad i_4 = 1\ \si{\ampere}, \quad i_5 = -4\ \si{\ampere}}
\end{equation}
\textbf{5) Cálculo de Potencias y Comprobación del Balance de Tellegen:}
\begin{itemize}[leftmargin=*]
    \item \textbf{Potencias disipadas en las resistencias ($P_R = I^2 R$):}
    \begin{align*}
        P_{R1} &= i_1^2 R_1 = 3^2 \cdot 2 = 18\ \si{\watt}, &
        P_{R2} &= i_2^2 R_2 = 2^2 \cdot 4 = 16\ \si{\watt}, &
        P_{R3} &= i_2^2 R_3 = 2^2 \cdot 1 = 4\ \si{\watt} \\
        P_{R4} &= i_5^2 R_4 = (-4)^2 \cdot 1 = 16\ \si{\watt}, &
        P_{R5} &= i_4^2 R_5 = 1^2 \cdot 8 = 8\ \si{\watt}
    \end{align*}
    \begin{equation}
        \sum P_{\text{resistencias}} = 18 + 16 + 4 + 16 + 8 = 62\ \si{\watt}
    \end{equation}
    \item \textbf{Potencias en las fuentes (criterio de generador):}
    \begin{itemize}
        \item Fuente $V_1 = 12\ \si{\volt}$: La corriente $i_1 = 3\ \si{\ampere}$ entra por su borne positivo, luego absorbe potencia:
        $P_{V1} = 12\ \si{\volt} \cdot (-3\ \si{\ampere}) = -36\ \si{\watt}$ (\textbf{absorbe $36\ \si{\watt}$}).
        \item Fuente $V_2 = 22\ \si{\volt}$: Como $i_3 = -5\ \si{\ampere}$, salen $5\ \si{\ampere}$ por su borne positivo, luego entrega potencia:
        $P_{V2} = 22\ \si{\volt} \cdot (-i_3) = 22 \cdot 5 = +110\ \si{\watt}$ (\textbf{entrega $110\ \si{\watt}$}).
        \item Fuente de corriente $I_s = 1\ \si{\ampere}$: Impulsa corriente desde $e$ ($8\ \si{\volt}$) hacia $d$ ($-4\ \si{\volt}$), por lo que:
        $P_{Is} = (V_d - V_e) \cdot I_s = (-4 - 8) \cdot 1 = -12\ \si{\watt}$ (\textbf{absorbe $12\ \si{\watt}$}).
    \end{itemize}
    \item \textbf{Verificación del Balance de Potencias:}
    \begin{equation}
        P_{\text{neta fuentes}} = +110\ \si{\watt} - 36\ \si{\watt} - 12\ \si{\watt} = 62\ \si{\watt} = \sum P_{\text{resistencias}} \quad \text{[OK]}
    \end{equation}
\end{itemize}
\textit{(Nota complementaria: en la variante alternativa donde el punto intermedio $c$ se toma con $V_c = 14\ \si{\volt}$, $R_3 = 5\ \si{\ohm}$ entre $b$ y $e$ y $R_4 = 10\ \si{\ohm}$, las corrientes son $I_1 = 2\ \si{\ampere}, I_2 = 1{,}5\ \si{\ampere}, I_3 = 2\ \si{\ampere}, I_4 = -0{,}4\ \si{\ampere}, I_5 = 1\ \si{\ampere}$ y el balance resulta igualmente exacto con $\sum P_R = 8 + 9 + 20 + 1{,}6 + 8 = 46{,}6\ \si{\watt}$ y $P_{\text{suministrada}} = 55\ \si{\watt} = 46{,}6 + 7{,}2 + 1{,}2\ \si{\watt}$).}
\end{ejercicio}

\begin{ejercicio}[Parcial 1 --- Problema 3: Circuito AC por Nudos, Superposición y Transformación de Fuentes]
Determinar los fasores de tensión $\mathbf{V}_1$ y $\mathbf{V}_2$ en el circuito de corriente alterna de la figura (con $\mathbf{I}_1 = 20\angle 0^\circ\ \si{\milli\ampere} = 0{,}02\ \si{\ampere}$, $\mathbf{I}_2 = 50\angle -90^\circ\ \si{\milli\ampere} = -j0{,}05\ \si{\ampere}$, $\mathbf{Z}_C = -j20\ \si{\ohm}$, $\mathbf{Z}_L = j40\ \si{\ohm}$ y $\mathbf{Z}_R = 25\ \si{\ohm}$) utilizando los tres métodos siguientes:
\begin{enumerate}[label=\alph*)]
    \item Resolución por el método de los nudos.
    \item Resolución por el teorema de superposición.
    \item Resolución por el método de transformación de fuentes.

    Y = (A + B) \cdot C + D \cdot E \qquad \left(\text{así como la variante invertida } Y' = \overline{(A + B) \cdot C + D}\right)
\end{equation}
empleando:
\begin{enumerate}[label=\alph*)]
    \item Lógica NMOS con carga activa NMOS de deplexión.
    \item Lógica CMOS estática complementaria completa.
\end{enumerate}
\tcbline
\textbf{a) Diseño con Lógica NMOS con Carga NMOS de Deplexión:}\\[4pt]
En lógica NMOS, la red de conmutación inferior (\textit{Pull-Down Network}, PDN) sintetiza directamente la expresión booleana sin negar, produciendo en su drenador común la función complementada $Y' = \overline{(A+B)\cdot C + D\cdot E}$:
\begin{itemize}[leftmargin=*]
    \item La suma lógica $(A + B)$ se implementa con dos transistores NMOS de enriquecimiento ($N_A, N_B$) en \textbf{paralelo}.
    \item El producto por $C$ conecta el bloque paralelo $(N_A \parallel N_B)$ en \textbf{serie} con el transistor $N_C$.
    \item La suma con el término $D \cdot E$ añade en \textbf{paralelo} una segunda rama vertical formada por $N_D$ y $N_E$ en \textbf{serie} (o un único transistor $N_D$ en la variante $Y' = \overline{(A+B)\cdot C + D}$).
    \item La carga de \textit{Pull-Up} hacia $V_{DD}$ es un transistor NMOS de deplexión con su puerta unida a su fuente ($V_{GS} = 0$).
    \item Para obtener la salida sin negar $Y = (A+B)\cdot C + D\cdot E$, se añade a continuación una etapa inversora NMOS.
\end{itemize}

\textbf{b) Diseño con Lógica CMOS Estática Complementaria:}\\[4pt]
En tecnología CMOS se elimina la carga resistiva o de deplexión y se emplea una \textbf{Red de Pull-Up (PUN)} formada exclusivamente por transistores PMOS que es el \textbf{grafo dual exacto} de la \textbf{Red de Pull-Down (PDN)} de transistores NMOS:
\begin{itemize}[leftmargin=*]
    \item \textbf{Red PDN (NMOS inferior):} Idéntica al apartado anterior: $\left[(N_A \parallel N_B) \text{ serie } N_C\right] \parallel (N_D \text{ serie } N_E)$.
    \item \textbf{Red PUN (PMOS superior):} Por dualidad serie $\leftrightarrow$ paralelo:
    \begin{itemize}
        \item El bloque $(D \cdot E)$ pasa a ser dos PMOS en \textbf{paralelo}: $(P_D \parallel P_E)$.
        \item El bloque $(A + B) \cdot C$ pasa a ser $(P_A \text{ en serie con } P_B)$ conectado en \textbf{paralelo} con $P_C$.
        \item Ambos subbloques se conectan en \textbf{serie} entre $V_{DD}$ y el nodo de salida $Y'$.
    \end{itemize}
    \item Finalmente, un \textbf{inversor CMOS} ($P_{inv}$ arriba y $N_{inv}$ abajo) invierte $Y'$ para entregar $Y = (A+B)C + DE$.
\end{itemize}

\begin{equation}
    \boxed{V_o = I \cdot 6\ \si{\kilo\ohm} = 0{,}93\ \si{\milli\ampere} \times 6\ \si{\kilo\ohm} = 5{,}58\ \si{\volt}}
\end{equation}
3) \textbf{Comprobación de las condiciones de consistencia:}
\begin{itemize}[leftmargin=*]
    \item Para $D_1$ (supuesto ON): $I_{D1} = 0{,}93\ \si{\milli\ampere} > 0$ (y el potencial antes del diodo es $10 - 4 \cdot 0{,}93 = 6{,}28\ \si{\volt}$, luego $V_{D1} = 6{,}28 - 5{,}58 = 0{,}7\ \si{\volt}$) $\implies$ \textbf{$D_1$ ON verificado}.
    \item Para $D_2$ (supuesto OFF): $V_{D2} = V_{A2} - V_o = 3\ \si{\volt} - 5{,}58\ \si{\volt} = -2{,}58\ \si{\volt} < V_\gamma = 0{,}7\ \si{\volt}$ $\implies$ \textbf{$D_2$ OFF verificado}.
\end{itemize}
4) \textbf{Comprobación adicional de que NO puede darse $D_1$ ON y $D_2$ ON:}
Si ambos condujeran, la rama de $D_2$ fijaría $V_o = 3\ \si{\volt} - 0{,}7\ \si{\volt} = 2{,}3\ \si{\volt}$. Entonces:
\begin{equation}
    I_{6\text{k}} = \frac{2{,}3\ \si{\volt}}{6\ \si{\kilo\ohm}} = 0{,}383\ \si{\milli\ampere}, \qquad I_{D1} = \frac{10 - 0{,}7 - 2{,}3}{4\ \si{\kilo\ohm}} = \frac{7\ \si{\volt}}{4\ \si{\kilo\ohm}} = 1{,}75\ \si{\milli\ampere}
\end{equation}
Aplicando LCK en el nudo $V_o$:
\begin{equation}
    I_{D1} + I_{D2} = I_{6\text{k}} \implies I_{D2} = I_{6\text{k}} - I_{D1} = 0{,}383 - 1{,}75 = -1{,}367\ \si{\milli\ampere} < 0
\end{equation}
Como un diodo no puede conducir corriente negativa de cátodo a ánodo, la hipótesis de conducción simultánea queda descartada.
\end{ejercicio}

\begin{ejercicio}[Parcial 2 --- Problema 2b: Discriminación de Estados ON/OFF en Red con Fuente de Corriente ($D_3, D_4$)]
Determinar razonadamente el estado de conducción (ON u OFF) de los diodos $D_3$ y $D_4$ y el valor de la tensión $V_o$ en el circuito de la figura, empleando el segundo modelo lineal del diodo con $V_\gamma = 0{,}7\ \si{\volt}$.

\textbf{Solución Paso a Paso (evaluando sistemáticamente las hipótesis del manuscrito `orig-119.png`):}\\[4pt]
1) \textbf{Análisis inicial:} La fuente de corriente de $5\ \si{\milli\ampere}$ inyecta corriente de forma obligatoria hacia el nudo $V_o$, por lo que no pueden estar ambos diodos en corte simultáneamente.\\[4pt]
2) \textbf{Hipótesis A ($D_3$ OFF y $D_4$ ON):}
Si $D_3$ está abierto (OFF), toda la corriente de la fuente ($I = 5\ \si{\milli\ampere}$) circula hacia la derecha por la resistencia de $1\ \si{\kilo\ohm}$ y regresa a tierra atravesando la fuente de $10\ \si{\volt}$ y el diodo $D_4$ ($I_{D4} = 5\ \si{\milli\ampere} > 0$).
Con la orientación mostrada en el esquema del examen (`orig-119.png`), el potencial en el nudo intermedio derecho es:
\begin{equation}
    V_x = 0 + 0{,}7\ \si{\volt} - 10\ \si{\volt} = -9{,}3\ \si{\volt}
\end{equation}
Y la tensión en el nudo $V_o$ resulta:
\begin{equation}
    V_o - V_x = I \cdot 1\ \si{\kilo\ohm} \implies V_o = -9{,}3\ \si{\volt} + (5\ \si{\milli\ampere} \times 1\ \si{\kilo\ohm}) = -9{,}3 + 5 = -4{,}3\ \si{\volt}
\end{equation}
Comprobamos el estado de $D_3$: al estar su cátodo a tierra ($0\ \si{\volt}$) y la tensión superior en $V_o = -4{,}3\ \si{\volt} < 0{,}7\ \si{\volt}$, el diodo $D_3$ tiene tensión ánodo-cátodo negativa ($V_{D3} < V_\gamma$), confirmando que \textbf{$D_3$ está OFF y $D_4$ está ON} con $V_o = -4{,}3\ \si{\volt}$.\\[4pt]
3) \textbf{Comprobación de las demás combinaciones ($D_3$ ON y $D_4$ ON, o $D_3$ ON y $D_4$ OFF):}
\begin{itemize}[leftmargin=*]
    \item Si suponemos \textbf{$D_3$ ON y $D_4$ ON}: entonces $V_x = -10{,}7\ \si{\volt}$ y la ecuación de nudo en $V_o$ es:
    \begin{equation}
        5\ \si{\milli\ampere} = \frac{V_o - 0{,}7}{2\ \si{\kilo\ohm}} + \frac{V_o + 10{,}7}{1\ \si{\kilo\ohm}} \implies 3 V_o = -10{,}7 \implies V_o = -3{,}57\ \si{\volt}
    \end{equation}
    Con $V_o = -3{,}57\ \si{\volt} < 0{,}7\ \si{\volt}$, por la rama de $D_3$ la corriente iría hacia arriba ($I_{D3} < 0$), lo cual contradice que $D_3$ esté ON.
    \item Si suponemos \textbf{$D_3$ ON y $D_4$ OFF}: toda la corriente de $5\ \si{\milli\ampere}$ bajaría por la rama de $2\ \si{\kilo\ohm}$, dando $V_o = 0{,}7 + 5\ \si{\milli\ampere} \times 2\ \si{\kilo\ohm} = +10{,}7\ \si{\volt}$, lo que polarizaría fuertemente en directa a la rama derecha haciendo conducir a $D_4$.
\end{itemize}
\end{ejercicio}

\begin{ejercicio}[Parcial 2 --- Problema 3: Circuito con Dos Amplificadores Operacionales Acoplados ($V_{AB}$)]
En el circuito de la figura, calcular las tensiones en los nodos de salida $V_A$ y $V_B$ de los dos amplificadores operacionales ideales $AO_1$ y $AO_2$, así como la diferencia de potencial entre ambos, $V_{AB} = V_A - V_B$.
\begin{itemize}[leftmargin=*]
    \item El operacional $AO_1$ tiene una resistencia de realimentación de $50\ \si{\kilo\ohm}$, su terminal no inversor conectado a tierra a través de $1\ \si{\mega\ohm}$, y a su entrada inversora llega una red formada por una fuente de corriente de $2\ \si{\milli\ampere}$ (dirigida hacia abajo) en paralelo con una resistencia de $1\ \si{\kilo\ohm}$, conectada en serie con una resistencia de $19\ \si{\kilo\ohm}$ a tierra.
    \item El operacional $AO_2$ está configurado como inversor con entrada de $0{,}5\ \si{\volt}$ a través de $1\ \si{\kilo\ohm}$ y resistencia de realimentación de $10\ \si{\kilo\ohm}$ (o equivalentemente $10\ \si{\kilo\ohm}$ y $100\ \si{\kilo\ohm}$, relación de ganancia $-10$), con una resistencia de acoplamiento de $10\ \si{\kilo\ohm}$ entre los nodos $A$ y $B$.

\textbf{Solución Paso a Paso:}\\[4pt]
1) \textbf{Cálculo de $V_B$ en el operacional $AO_2$:}
El operacional $AO_2$ forma un amplificador inversor cuya tensión de salida en el nodo $B$ queda fijada directamente por su lazo de realimentación negativa ($V^+ = V^- = 0\ \si{\volt}$):
\begin{equation}
    \boxed{V_B = -\frac{R_{f2}}{R_1} \cdot 0{,}5\ \si{\volt} = -\frac{10\ \si{\kilo\ohm}}{1\ \si{\kilo\ohm}} \cdot 0{,}5\ \si{\volt} = -5{,}0\ \si{\volt}}
\end{equation}
2) \textbf{Transformación de la fuente de corriente en la entrada de $AO_1$:}
La fuente de intensidad de $2\ \si{\milli\ampere}$ (apuntando hacia abajo) en paralelo con la resistencia de $1\ \si{\kilo\ohm}$ se transforma en su equivalente de Thévenin: una fuente de tensión de valor $V_s = 2\ \si{\milli\ampere} \times 1\ \si{\kilo\ohm} = 2\ \si{\volt}$ (con el polo negativo arriba y el positivo abajo) en serie con la resistencia de $1\ \si{\kilo\ohm}$.
Al sumarse en serie con la resistencia de $19\ \si{\kilo\ohm}$ conectada a tierra, la rama de entrada equivale a un generador de $-2\ \si{\volt}$ en serie con una resistencia total $R_{eq} = 1\ \si{\kilo\ohm} + 19\ \si{\kilo\ohm} = 20\ \si{\kilo\ohm}$.\\[4pt]
3) \textbf{Cálculo de $V_A$ en el operacional $AO_1$:}
Como no circula corriente por la resistencia de $1\ \si{\mega\ohm}$ ($I^+ = 0$), el terminal no inversor está a $V^+ = 0\ \si{\volt}$ y, por cortocircuito virtual, $V^- = V^+ = 0\ \si{\volt}$.
La intensidad $I_1$ que circula desde el generador equivalente hacia el nodo inversor es:
\begin{equation}
    I_1 = \frac{-2\ \si{\volt} - V^-}{20\ \si{\kilo\ohm}} = \frac{-2\ \si{\volt}}{20\ \si{\kilo\ohm}} = -0{,}1\ \si{\milli\ampere}
\end{equation}
Por la primera ley de Kirchhoff en el nodo inversor ($I^- = 0 \implies I_1 = I_2$):
\begin{equation}
    I_2 = \frac{0 - V_A}{50\ \si{\kilo\ohm}} = I_1 \implies \boxed{V_A = -50\ \si{\kilo\ohm} \cdot I_1 = -50\ \si{\kilo\ohm} \cdot (-0{,}1\ \si{\milli\ampere}) = +5{,}0\ \si{\volt}}
\end{equation}
4) \textbf{Diferencia de potencial $V_{AB}$:}
\begin{equation}
    \boxed{V_{AB} = V_A - V_B = 5{,}0\ \si{\volt} - (-5{,}0\ \si{\volt}) = +10{,}0\ \si{\volt}}
\end{equation}
\end{ejercicio}
