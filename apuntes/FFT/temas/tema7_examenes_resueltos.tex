\section{Examen Parcial 1: Electromagnetismo y Circuitos DC/AC}

\begin{ejercicio}[Parcial 1 --- Problema 1: Campo Magnético y Fuerza entre Tres Hilos Paralelos]
Se tienen tres cables conductores rectilíneos, paralelos e indefinidos de longitud $L = 50\ \si{\centi\meter} = 0{,}5\ \si{\meter}$, situados coplanariamente en el vacío ($\mu_0 = 4\pi \times 10^{-7}\ \si{\tesla\meter\per\ampere}$).
Por los tres hilos circulan corrientes continuas en el mismo sentido ($+\hat{k}$ saliendo del plano):
\begin{equation}
    I_1 = 2\ \si{\milli\ampere}, \qquad I_2 = 4\ \si{\milli\ampere}, \qquad I_3 = 6\ \si{\milli\ampere}
\end{equation}
La distancia entre los cables 1 y 2 es $d_{12} = 1\ \si{\centi\meter}$, y entre los cables 2 y 3 es $d_{23} = 2\ \si{\centi\meter}$.
Calcular los campos magnéticos y las fuerzas producidas sobre cada hilo.
\tcbline
\textbf{Diagrama Geométrico del Problema en Corte Transversal:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=1.1]
    % Eje horizontal
    \draw[->, gray] (-0.8,0) -- (4.5,0) node[right, text=black] {$x$};
    % Hilo 1 (x = 0)
    \draw[thick, fftPrimary] (0,0) circle (0.28);
    \filldraw[fftPrimary] (0,0) circle (2.5pt);
    \node[below=8pt] at (0,0) {$H_1\ (\odot\ I_1{=}2\si{\milli\ampere})$};
    % Hilo 2 (x = 1 cm)
    \draw[thick, fftDarkBlue] (1.5,0) circle (0.28);
    \filldraw[fftDarkBlue] (1.5,0) circle (2.5pt);
    \node[below=8pt] at (1.5,0) {$H_2\ (\odot\ I_2{=}4\si{\milli\ampere})$};
    % Hilo 3 (x = 3 cm)
    \draw[thick, fftGreen] (3.8,0) circle (0.28);
    \filldraw[fftGreen] (3.8,0) circle (2.5pt);
    \node[below=8pt] at (3.8,0) {$H_3\ (\odot\ I_3{=}6\si{\milli\ampere})$};
    % Cotas de separación
    \draw[<->, gray] (0,0.8) -- (1.5,0.8) node[midway, above] {$d_{12}{=}1\si{\centi\meter}$};
    \draw[<->, gray] (1.5,0.8) -- (3.8,0.8) node[midway, above] {$d_{23}{=}2\si{\centi\meter}$};
    % Vectores de fuerza neta
    \draw[->, >=stealth, very thick, fftAccent] (0,-0.6) -- (0.9,-0.6) node[right] {$\vec{F}_1$};
    \draw[->, >=stealth, very thick, fftAccent] (1.5,-0.6) -- (0.8,-0.6) node[left] {$\vec{F}_2$};
    \draw[->, >=stealth, very thick, fftAccent] (3.8,-0.6) -- (2.6,-0.6) node[left] {$\vec{F}_3$};
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
a) Aplicando la Ley de Ampère: $B(r) = \frac{\mu_0 I}{2\pi r}$.\\[4pt]
b) \textbf{Campos magnéticos sobre cada cable:}\\[2pt]
- Sobre $H_1$: $B_1 = B_{2 \to 1} + B_{3 \to 1} = (8 + 4) \times 10^{-8} = 12 \times 10^{-8}\ \si{\tesla}\ (+\hat{j})$.\\[2pt]
- Sobre $H_2$: $B_2 = -B_{1 \to 2} + B_{3 \to 2} = (-4 + 6) \times 10^{-8} = +2 \times 10^{-8}\ \si{\tesla}\ (+\hat{j})$.\\[2pt]
- Sobre $H_3$: $B_3 = -(B_{1 \to 3} + B_{2 \to 3}) = -(1{,}333 + 4) \times 10^{-8} = -5{,}333 \times 10^{-8}\ \si{\tesla}\ (\hat{j})$.\\[4pt]
c) \textbf{Fuerzas magnéticas netas ($\vec{F} = I \vec{L} \times \vec{B}$):}
\begin{align}
    \vec{F}_1 &= -1{,}2 \times 10^{-10}\ \si{\newton}\ \hat{i} \quad (\text{hacia la derecha}) \\
    \vec{F}_2 &= -0{,}4 \times 10^{-10}\ \si{\newton}\ \hat{i} \quad (\text{hacia la derecha}) \\
    \vec{F}_3 &= +1{,}6 \times 10^{-10}\ \si{\newton}\ \hat{i} \quad (\text{hacia la izquierda})
\end{align}
\begin{equation}
    \sum \vec{F} = (-1{,}2 - 0{,}4 + 1{,}6) \times 10^{-10}\ \hat{i} = \vec{0} \quad \text{[Tercera Ley de Newton Verificada]}
\end{equation}
\end{ejercicio}

\begin{ejercicio}[Parcial 1 --- Problema 2: Análisis por Nudos y Balance de Potencia en DC]
Dado un circuito de corriente continua con $V_1 = 12\ \si{\volt}, V_2 = 22\ \si{\volt}$, $I_s = 1\ \si{\ampere}$, $R_1 = 2\ \si{\ohm}, R_2 = 4\ \si{\ohm}, R_3 = 5\ \si{\ohm}, R_4 = 10\ \si{\ohm}, R_5 = 8\ \si{\ohm}$.
Tras formular el sistema de nudos se obtienen: $V_b = 18\ \si{\volt}, V_c = 14\ \si{\volt}, V_d = -4\ \si{\volt}, V_e = 8\ \si{\volt}$.
Calcular las corrientes, potencias disipadas y comprobar el balance de Tellegen.
\tcbline
\textbf{Esquema Circuital con Nudos y Corrientes:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    % Nudo de masa
    \draw (0,0) node[ground] {} node[below] {$V_f{=}0\si{\volt}$};
    \draw (-3,0) -- (3,0);
    % Ramas
    \draw (-3,0) to[V, v=$V_1{=}12\si{\volt}$] (-3,2.5) node[above] {$V_a$};
    \draw (-3,2.5) to[R, l=$R_1{=}2\si{\ohm}$, i=$I_1$] (0,2.5) node[above] {$V_b{=}18\si{\volt}$};
    \draw (0,2.5) to[R, l=$R_3{=}5\si{\ohm}$, i=$I_3$] (0,0);
    \draw (0,2.5) to[V, v=$V_2{=}22\si{\volt}$] (3,2.5) node[above] {$V_c{=}14\si{\volt}$};
    \draw (3,2.5) to[R, l=$R_2{=}4\si{\ohm}$, i=$I_2$] (3,0);
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
1) Corrientes de rama:
\begin{align}
    I_1 &= \frac{V_b - V_c}{R_1} = \frac{18 - 14}{2} = 2\ \si{\ampere} \\
    I_2 &= \frac{V_c - V_e}{R_2} = \frac{14 - 8}{4} = 1{,}5\ \si{\ampere} \\
    I_3 &= \frac{V_b - V_e}{R_3} = \frac{18 - 8}{5} = 2\ \si{\ampere}
\end{align}
2) Potencias disipadas: $P_{R1} = 2^2 \times 2 = 8\ \si{\watt}$, $P_{R2} = 1{,}5^2 \times 4 = 9\ \si{\watt}$, $P_{R3} = 2^2 \times 5 = 20\ \si{\watt}$, $P_{R4} = 1{,}6\ \si{\watt}$, $P_{R5} = 8\ \si{\watt}$. Total: $\sum P_R = 46{,}6\ \si{\watt}$.\\
3) Potencias en fuentes: $P_{V2,entregada} = 22 \times 2{,}5 = 55\ \si{\watt}$; fuentes que absorben: $P_{V1,abs} = 7{,}2\ \si{\watt}$, $P_{Is,abs} = 1{,}2\ \si{\watt}$.\\
4) Balance global:
\begin{equation}
    P_{suministrada} - P_{absorbida} = 55{,}0 - (46{,}6 + 7{,}2 + 1{,}2) = 55{,}0 - 55{,}0 = 0\ \si{\watt} \quad \text{[CORRECTO]}
\end{equation}
\end{ejercicio}

\begin{ejercicio}[Parcial 1 --- Problema 3: Circuito AC por Transformación de Fuentes]
Determinar los fasores de tensión en el circuito AC con $\mathbf{I}_1 = 20\angle 0^\circ\ \si{\milli\ampere}$, $\mathbf{I}_2 = 50\angle -90^\circ\ \si{\milli\ampere} = -j0{,}05\ \si{\ampere}$, $\mathbf{Z}_C = -j20\ \si{\ohm}$, $\mathbf{Z}_L = j40\ \si{\ohm}$ y $\mathbf{Z}_R = 25\ \si{\ohm}$.
\tcbline
\textbf{Paso 1: Esquema Circuital Original en Alterna:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    % Fuente I1 y ZC
    \draw (0,0) to[I, i=$\mathbf{I}_1{=}0{,}02\si{\ampere}$] (0,2.5) -- (2,2.5);
    \draw (2,2.5) to[C, l=$\mathbf{Z}_C{=}-j20\si{\ohm}$] (2,0) -- (0,0);
    % Resistencia ZR en serie
    \draw (2,2.5) to[R, l=$\mathbf{Z}_R{=}25\si{\ohm}$] (4.5,2.5);
    % ZL e I2
    \draw (4.5,2.5) to[L, l=$\mathbf{Z}_L{=}j40\si{\ohm}$] (4.5,0) -- (2,0);
    \draw (4.5,2.5) -- (6.5,2.5) to[I, i=$\mathbf{I}_2{=}-j0{,}05\si{\ampere}$] (6.5,0) -- (4.5,0);
\end{tikzpicture}
\end{center}

\textbf{Paso 2: Circuito Transformado a Fuentes de Tensión Serie:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    \draw (0,0) to[sV, v=$\mathbf{V}_{s1}{=}-j0{,}4\si{\volt}$] (0,2)
          to[C, l=$\mathbf{Z}_C{=}-j20\si{\ohm}$] (2.2,2)
          to[R, l=$\mathbf{Z}_R{=}25\si{\ohm}$] (4.4,2)
          to[L, l=$\mathbf{Z}_L{=}j40\si{\ohm}$] (6.6,2)
          to[sV, v=$\mathbf{V}_{s2}{=}2\si{\volt}$] (6.6,0) -- (0,0);
    \draw[->, >=stealth, very thick, fftAccent] (2.2,2.3) -- (3.2,2.3) node[midway, above] {$\mathbf{I}_{lazo}$};
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
1) Transformación de Thévenin:
\begin{align}
    \mathbf{V}_{s1} &= \mathbf{I}_1 \cdot \mathbf{Z}_C = 0{,}02 \times (-j20) = -j0{,}4\ \si{\volt} \\
    \mathbf{V}_{s2} &= \mathbf{I}_2 \cdot \mathbf{Z}_L = (-j0{,}05) \times (j40) = +2\ \si{\volt}
\end{align}
2) Impedancia total y corriente de lazo:
\begin{equation}
    \mathbf{Z}_{total} = 25 + j(40 - 20) = 25 + j20\ \si{\ohm}
\end{equation}
\begin{equation}
    \mathbf{I}_{lazo} = \frac{\mathbf{V}_{s2} - \mathbf{V}_{s1}}{\mathbf{Z}_{total}} = \frac{2 + j0{,}4}{25 + j20} \approx 0{,}0566 - j0{,}0293\ \si{\ampere}
\end{equation}
3) Tensión en la resistencia:
\begin{equation}
    \mathbf{V}_2 = \mathbf{I}_{lazo} \cdot \mathbf{Z}_R \approx (0{,}0566 - j0{,}0293) \times 25 \approx 1{,}415 - j0{,}732\ \si{\volt} = 1{,}59 \angle -27{,}3^\circ\ \si{\volt}
\end{equation}
\end{ejercicio}

\section{Examen Parcial 2: Dispositivos Electrónicos y Electrónica Digital}

\begin{ejercicio}[Parcial 2 --- Problema 1: Diodos ¿ON u OFF? --- Análisis Sistemático]
Determinar el estado de conducción de los diodos $D_1$ y $D_2$ bajo el modelo $V_\gamma = 0{,}7\ \si{\volt}$.
$D_1$ está alimentado por $10\ \si{\volt}$ con $R_1 = 4\ \si{\kilo\ohm}$. $D_2$ está alimentado por $3\ \si{\volt}$. Ambos cátodos se unen al nudo $V_x$ conectado a masa mediante $R_3 = 6\ \si{\kilo\ohm}$.
\tcbline
\textbf{Esquema Circuital con Diodos:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    % Rama D1
    \draw (0,0) to[V, v=$10\si{\volt}$] (0,2.5)
          to[R, l=$R_1{=}4\si{\kilo\ohm}$] (2.2,2.5)
          to[D, l=$D_1$] (4,2.5) -- (4,1.5);
    % Rama D2
    \draw (1.5,0) to[V, v=$3\si{\volt}$] (1.5,1.5)
          to[D, l=$D_2$] (4,1.5) node[right=4pt] {$V_x$};
    % Resistencia de carga a masa
    \draw (4,1.5) to[R, l=$R_3{=}6\si{\kilo\ohm}$, i=$I_3$] (4,-0.5) node[ground] {};
    \draw (0,0) -- (1.5,0);
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
1) \textbf{Hipótesis: $D_1$ ON y $D_2$ OFF:}\\
Sustituyendo $D_1$ por su batería de $V_\gamma = 0{,}7\ \si{\volt}$ y $D_2$ por circuito abierto:
\begin{equation}
    I_{D1} = \frac{10 - 0{,}7}{4\ \si{\kilo\ohm} + 6\ \si{\kilo\ohm}} = \frac{9{,}3}{10} = 0{,}93\ \si{\milli\ampere}
\end{equation}
\begin{equation}
    V_x = I_{D1} \cdot R_3 = 0{,}93 \times 6 = 5{,}58\ \si{\volt}
\end{equation}
2) \textbf{Verificación:}\\[2pt]
- $D_1$ (ON): $I_{D1} = 0{,}93\ \si{\milli\ampere} > 0$ \quad [VERIFICADO]\\
- $D_2$ (OFF): $V_{D2} = 3 - V_x = 3 - 5{,}58 = -2{,}58\ \si{\volt} < 0{,}7\ \si{\volt}$ \quad [VERIFICADO: bloqueado en inversa]\\[2pt]
\textbf{Resultado:} $D_1$ está ON ($I = 0{,}93\ \si{\milli\ampere}$), $D_2$ está OFF ($V_{D2} = -2{,}58\ \si{\volt}$) y $V_x = 5{,}58\ \si{\volt}$.
\end{ejercicio}

\begin{ejercicio}[Parcial 2 --- Problema 2: Dos Amplificadores Operacionales Acoplados]
Calcular la tensión diferencial $V_{AB} = V_A - V_B$ en el circuito compuesto por dos operacionales ideales acoplados: $OA_1$ con $I_{in} = -0{,}1\ \si{\milli\ampere}$ y $R_{f1} = 50\ \si{\kilo\ohm}$; $OA_2$ con $V_{ref} = 0{,}5\ \si{\volt}$, $R_1 = 10\ \si{\kilo\ohm}$ y $R_{f2} = 100\ \si{\kilo\ohm}$.
\tcbline
\textbf{Esquema Circuital de las Dos Etapas con Op-Amps:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=0.95]
    % Etapa 1 (OA1)
    \node[font=\bfseries] at (-3,3.4) {Etapa 1 ($OA_1$)};
    \draw (-4.5,1.5) node[op amp] (oa1) {};
    \draw (oa1.+) -- (-5.5,1.0) node[ground] {};
    \draw (oa1.-) -- (-5.5,2.0) to[short, i<=$I_{in}{=}-0{,}1\si{\milli\ampere}$] (-7,2.0);
    \draw (oa1.-) -- (-5.5,2.8) to[R, l=$R_{f1}{=}50\si{\kilo\ohm}$] (-3.5,2.8) -- (-3.5,1.5) -- (oa1.out);
    \draw (oa1.out) to[short, -o] (-2.2,1.5) node[right] {$V_A$};

    % Etapa 2 (OA2)
    \node[font=\bfseries] at (3.5,3.4) {Etapa 2 ($OA_2$)};
    \draw (2.5,1.5) node[op amp] (oa2) {};
    \draw (oa2.+) -- (1.5,1.0) node[ground] {};
    \draw (0.2,2.0) node[left] {$V_{ref}{=}0{,}5\si{\volt}$} to[R, l=$R_1{=}10\si{\kilo\ohm}$] (oa2.-);
    \draw (oa2.-) -- (1.5,2.8) to[R, l=$R_{f2}{=}100\si{\kilo\ohm}$] (3.5,2.8) -- (3.5,1.5) -- (oa2.out);
    \draw (oa2.out) to[short, -o] (4.8,1.5) node[right] {$V_B$};
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
1) Etapa 1 ($OA_1$ con masa virtual $V^- = 0$):
\begin{equation}
    I_{in} = \frac{0 - V_A}{R_{f1}} \implies -0{,}1\ \si{\milli\ampere} = -\frac{V_A}{50\ \si{\kilo\ohm}} \implies V_A = +5{,}0\ \si{\volt}
\end{equation}
2) Etapa 2 ($OA_2$ inversor):
\begin{equation}
    V_B = -\frac{R_{f2}}{R_1} V_{ref} = -\frac{100\ \si{\kilo\ohm}}{10\ \si{\kilo\ohm}} \times (0{,}5\ \si{\volt}) = -5{,}0\ \si{\volt}
\end{equation}
3) Tensión diferencial:
\begin{equation}
    V_{AB} = V_A - V_B = 5{,}0 - (-5{,}0) = 10{,}0\ \si{\volt}
\end{equation}
\end{ejercicio}
