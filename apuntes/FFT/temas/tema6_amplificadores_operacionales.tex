\section{El Amplificador Operacional y el Cortocircuito Virtual}

\subsection{Símbolo Circuital y Reglas de Oro}
\begin{definicion}[Características del Op-Amp Ideal]
\begin{enumerate}[leftmargin=*]
    \item Ganancia diferencial infinita: $A_{vd} \to \infty$.
    \item Impedancia de entrada infinita: $R_{in} \to \infty \implies \mathbf{I^+ = 0, \quad I^- = 0}$.
    \item Impedancia de salida nula: $R_{out} = 0$.
    \item \textbf{Principio del Cortocircuito Virtual (con realimentación negativa):} $\mathbf{V^+ = V^-}$.
\end{enumerate}
\end{definicion}

\begin{center}
\begin{tikzpicture}[scale=1.0]
    \draw (0,0) node[op amp] (opamp) {};
    \node[left] at (opamp.-) {$V^-$ (Inversora, $I^- = 0$)};
    \node[left] at (opamp.+) {$V^+$ (No inversora, $I^+ = 0$)};
    \node[right] at (opamp.out) {$V_o$};
    \draw (opamp.up) -- (0,1.2) node[above] {$+V_{CC}$};
    \draw (opamp.down) -- (0,-1.2) node[below] {$-V_{EE}$};
    \node[below=16pt] at (0,-1.2) {\small\textsf{Símbolo circuital del Amplificador Operacional (Op-Amp)}};
\end{tikzpicture}
\end{center}

\section{Ejercicios Resueltos de Exámenes de Amplificadores Operacionales}

\begin{ejercicio}[Circuito Sumador-Inversor con Chequeo de Saturación --- Examen UGR]
Un amplificador operacional ideal con realimentación negativa dispone de $R_f = 20\ \si{\kilo\ohm}$ y saturación $\pm 14\ \si{\volt}$.
Entradas: $V_1 = +2\ \si{\volt}$ por $R_1 = 10\ \si{\kilo\ohm}$ y $V_2 = -3\ \si{\volt}$ por $R_2 = 5\ \si{\kilo\ohm}$.
Hallar $V_o$ y determinar si satura si se cambia $V_2$ a $-6\ \si{\volt}$.
\tcbline
\textbf{Esquema Circuital del Sumador Inversor:}\\[4pt]
\begin{center}
\begin{tikzpicture}[scale=1.0]
    \draw (0,1) node[op amp] (op) {};
    \draw (op.+) -- (-1,0.5) node[ground] {};
    \draw (op.-) -- (-1,1.5);
    \draw (-3,2.2) node[left] {$V_1{=}+2\si{\volt}$} to[R, l=$R_1{=}10\si{\kilo\ohm}$] (-1,1.5);
    \draw (-3,0.8) node[left] {$V_2$} to[R, l=$R_2{=}5\si{\kilo\ohm}$] (-1,1.5);
    \draw (-1,1.5) -- (-1,2.5) to[R, l=$R_f{=}20\si{\kilo\ohm}$] (1,2.5) -- (1,1) -- (op.out);
    \draw (op.out) -- (2,1) node[right] {$V_o$};
    \node[above] at (op.up) {$+15\si{\volt}$};
    \node[below] at (op.down) {$-15\si{\volt}$};
\end{tikzpicture}
\end{center}

\textbf{Solución Paso a Paso:}\\[4pt]
a) Con $V_1 = +2\ \si{\volt}$ y $V_2 = -3\ \si{\volt}$:
\begin{equation}
    V_o = -20 \left( \frac{2}{10} + \frac{-3}{5} \right) = -20 (0{,}20 - 0{,}60) = -20 (-0{,}40) = +8{,}0\ \si{\volt}
\end{equation}
Como $|V_o| = 8{,}0\ \si{\volt} < V_{sat} = 14\ \si{\volt}$, trabaja en régimen lineal sin saturación.\\[4pt]
b) Con $V_2 = -6\ \si{\volt}$:
\begin{equation}
    V_{o,calculado} = -20 \left( \frac{2}{10} + \frac{-6}{5} \right) = -20 (0{,}20 - 1{,}20) = -20 (-1{,}00) = +20{,}0\ \si{\volt}
\end{equation}
Como $20\ \si{\volt} > V_{sat} = +14\ \si{\volt}$, la salida satura en su límite físico positivo:
\begin{equation}
    V_o = +V_{sat} = +14{,}0\ \si{\volt}
\end{equation}
\end{ejercicio}

\begin{ejercicio}[Integrador Inversor ante Entrada Escalonada]
Un integrador inversor ideal dispone de $R = 50\ \si{\kilo\ohm}$, $C = 2\ \si{\micro\farad}$ y $V_{sat} = \pm 12\ \si{\volt}$.
En $t = 0$ se aplica una tensión escalón constante $V_i = +2\ \si{\volt}$ con $V_o(0) = 0$.
Hallar $V_o(t)$, la pendiente de la rampa y el tiempo hasta saturar.
\tcbline
\textbf{Solución Paso a Paso:}\\[4pt]
1) Constante de tiempo integradora:
\begin{equation}
    RC = (50 \times 10^3\ \si{\ohm}) \times (2 \times 10^{-6}\ \si{\farad}) = 0{,}10\ \si{\second} = 100\ \si{\milli\second}
\end{equation}
2) Salida lineal temporal:
\begin{equation}
    V_o(t) = -\frac{V_i}{RC} t = -\frac{2}{0{,}10} t = -20 t\ \si{\volt} \quad (\text{pendiente: } -20\ \si{\volt\per\second})
\end{equation}
3) Tiempo de saturación en $-V_{sat} = -12\ \si{\volt}$:
\begin{equation}
    -20 t_{sat} = -12 \implies t_{sat} = \frac{12}{20} = 0{,}60\ \si{\second} = 600\ \si{\milli\second}
\end{equation}
\end{ejercicio}
